/**
 * API pública do módulo `iniciativas`. Os outros módulos só falam com ele por aqui, e
 * sempre por portas implementadas em `infrastructure/adapters` do módulo consumidor.
 * É o único módulo que muda o status de uma iniciativa.
 */
export class IniciativasFacade {
  constructor(
    private readonly submeterIniciativa: SubmeterIniciativa,
    private readonly editarIniciativa: EditarIniciativa,
    private readonly ressubmeterIniciativa: RessubmeterIniciativa,
    private readonly decidirIniciativa: DecidirIniciativa,
    private readonly consultas: ConsultasIniciativas,
    private readonly carregar: CarregarComPrazos,
    private readonly job: AplicarPrazosVencidos,
    private readonly alterarDatasIniciativa: AlterarDatasIniciativa,
  ) {}

  // --- contrato usado pelos outros módulos --------------------------------------------------

  /**
   * Estado da iniciativa para validar operações de outros módulos (votar, comentar).
   * Aplica antes as transições por prazo vencidas (RN-06). `null` se não existir.
   */
  async obterEstado(iniciativaId: string): Promise<EstadoIniciativaDto | null> {
    const iniciativa = await this.carregar.executar(iniciativaId);
    if (!iniciativa) return null;
    return {
      id: iniciativa.id,
      autorId: iniciativa.autorId,
      status: iniciativa.status,
      ciclo: iniciativa.ciclo,
      comVotacao: iniciativa.comVotacao,
      fimRefinamento: iniciativa.fimRefinamento?.toISOString() ?? null,
      fimVotacao: iniciativa.fimVotacao?.toISOString() ?? null,
    };
  }

  /**
   * Registra a decisão do comitê (RN-10). Lança erros de domínio: `proibido` se o decisor
   * é o autor, `conflito` fora de Em análise ou se já foi decidida, `nao_encontrado` se
   * não existir. A validação da justificativa fica no caso de uso de `avaliacao`.
   */
  registrarDecisao(input: RegistrarDecisaoInput): Promise<void> {
    return this.decidirIniciativa.executar(
      input.iniciativaId,
      input.decisorId,
      input.resultado,
      input.justificativa,
    );
  }

  /** Listagem paginada de resumos, com filtros e ordenação no banco (RNF-05). */
  listar(filtro: FiltroListagemDto): Promise<PaginaDto<ResumoIniciativaDto>> {
    return this.consultas.listar(filtro);
  }

  /** Detalhe com permissões calculadas para o ator. Lança `nao_encontrado`. */
  detalhe(iniciativaId: string, ator: AtorDto): Promise<DetalheIniciativaDto> {
    return this.consultas.detalhe(iniciativaId, ator);
  }

  /** Iniciativas aprovadas cujo período pode cruzar o intervalo, para gerar ocorrências. */
  aprovadasParaCalendario(filtro: FiltroCalendarioInput): Promise<AprovadaParaCalendarioDto[]> {
    return this.consultas.aprovadasParaCalendario(filtro);
  }

  // --- usado pelos controllers e pelo job do próprio módulo ---------------------------------

  submeter(autorId: string, entrada: DadosIniciativaInput): Promise<string> {
    return this.submeterIniciativa.executar(autorId, entrada);
  }

  editar(id: string, usuarioId: string, entrada: DadosIniciativaInput): Promise<void> {
    return this.editarIniciativa.executar(id, usuarioId, entrada);
  }

  ressubmeter(id: string, usuarioId: string, entrada: DadosIniciativaInput): Promise<void> {
    return this.ressubmeterIniciativa.executar(id, usuarioId, entrada);
  }

  /** Comitê ou ADMIN muda datas e horários de uma aprovada (a série inteira). */
  alterarDatas(id: string, ator: AtorDto, datas: DatasIniciativaInput): Promise<void> {
    return this.alterarDatasIniciativa.executar(id, ator, datas);
  }

  minhas(autorId: string, filtro: { status?: StatusIniciativa; page: number; limit: number }) {
    return this.consultas.minhas(autorId, filtro);
  }

  contagens(): Promise<{ emAnalise: number }> {
    return this.consultas.contagens();
  }

  historico(iniciativaId: string): Promise<TransicaoDto[]> {
    return this.consultas.historico(iniciativaId);
  }

  /** Aplica as transições por prazo vencidas (job). Devolve quantas iniciativas processou. */
  aplicarPrazosVencidos(): Promise<number> {
    return this.job.executar();
  }
}

const iso = (data: Date | null): string | null => (data ? data.toISOString() : null);

const NOME_DESCONHECIDO = 'Membro da comunidade';

function autorDe(id: string, usuarios: Map<string, UsuarioResumo>): AutorDto {
  const usuario = usuarios.get(id);
  return {
    id,
    nome: usuario?.nome ?? NOME_DESCONHECIDO,
    perfil: usuario?.perfil ?? 'ALUNO',
  };
}

function prazoAtual(linha: LinhaResumo): Date | null {
  if (linha.status === 'EM_ANDAMENTO') return linha.fimRefinamento;
  if (linha.status === 'EM_VOTACAO') return linha.fimVotacao;
  return null;
}

function paraResumo(
  linha: LinhaResumo,
  usuarios: Map<string, UsuarioResumo>,
  sugestoes: Map<string, number>,
): ResumoIniciativaDto {
  return {
    id: linha.id,
    tipo: linha.tipo,
    titulo: linha.titulo,
    resumo: linha.resumo,
    autor: autorDe(linha.autorId, usuarios),
    status: linha.status,
    comVotacao: linha.comVotacao,
    ciclo: linha.ciclo,
    fimRefinamento: iso(linha.fimRefinamento),
    fimVotacao: iso(linha.fimVotacao),
    prazoAtual: iso(prazoAtual(linha)),
    votos: { likes: linha.likes, dislikes: linha.dislikes, relevancia: linha.relevancia },
    sugestoesCicloAtual: sugestoes.get(linha.id) ?? 0,
    inicio: linha.inicio,
    fim: linha.fim,
    periodicidade: linha.periodicidade,
    espaco: linha.espaco,
    criadaEm: linha.criadaEm.toISOString(),
    decididaEm: iso(linha.decididaEm),
  };
}

interface DepsConsultas {
  leitura: IniciativaLeituraPort;
  usuarios: UsuariosPort;
  sugestoes: SugestoesPort;
  votos: VotosPort;
  googleAgenda: GoogleAgendaPort;
  carregar: CarregarComPrazos;
  clock: Clock;
}

/** Casos de uso de leitura de `iniciativas` (listas, detalhe, histórico, calendário). */
export class ConsultasIniciativas {
  constructor(private readonly deps: DepsConsultas) {}

  /** RNF-05: filtros, ordenação e paginação no banco. */
  async listar(filtro: FiltroListagemDto): Promise<PaginaDto<ResumoIniciativaDto>> {
    const { linhas, total } = await this.deps.leitura.listar({
      status: filtro.status,
      tipo: filtro.tipo,
      q: filtro.q?.trim() || undefined,
      autorId: filtro.autorId,
      ordenar: filtro.ordenar ?? 'recentes',
      page: filtro.page,
      limit: filtro.limit,
    });
    return { items: await this.resumos(linhas), total };
  }

  /** Minhas iniciativas, com a contagem por status (todas, sem o filtro de status). */
  async minhas(
    autorId: string,
    filtro: { status?: StatusIniciativa; page: number; limit: number },
  ): Promise<
    PaginaDto<ResumoIniciativaDto> & {
      contagemPorStatus: Partial<Record<StatusIniciativa, number>>;
    }
  > {
    const [pagina, contagemPorStatus] = await Promise.all([
      this.listar({ ...filtro, autorId, ordenar: 'recentes' }),
      this.deps.leitura.contagemPorStatus(autorId),
    ]);
    return { ...pagina, contagemPorStatus };
  }

  async contagens(): Promise<{ emAnalise: number }> {
    return { emAnalise: await this.deps.leitura.contarPorStatus('EM_ANALISE') };
  }

  /** Detalhe com permissões calculadas no domínio. Aplica antes os prazos vencidos (RN-06). */
  async detalhe(id: string, ator: AtorDto): Promise<DetalheIniciativaDto> {
    const { carregar, leitura, usuarios, sugestoes, votos, googleAgenda, clock } = this.deps;
    const iniciativa = await carregar.executar(id);
    if (!iniciativa) throw new IniciativaNaoEncontradaError();
    const linha = await leitura.obterLinha(id);
    if (!linha) throw new IniciativaNaoEncontradaError();

    const ids = [linha.autorId, ...(linha.decisorId ? [linha.decisorId] : [])];
    const [mapaUsuarios, mapaSugestoes, meuVoto, ultimaJustificativa] = await Promise.all([
      usuarios.mapaPorIds(ids),
      sugestoes.contar([{ iniciativaId: linha.id, ciclo: linha.ciclo }]),
      votos.meuVoto(linha.id, linha.ciclo, ator.id),
      this.ultimaJustificativa(linha),
    ]);

    const decidida = linha.status === 'APROVADA' || linha.status === 'REPROVADA';
    return {
      ...paraResumo(linha, mapaUsuarios, mapaSugestoes),
      detalhe: linha.detalhe,
      horaInicio: linha.horaInicio,
      horaFim: linha.horaFim,
      organizadores: linha.organizadores,
      membros: linha.membros,
      participantes: linha.participantes,
      meuVoto,
      permissoes: iniciativa.permissoesPara(ator, clock),
      decisao:
        decidida && linha.decisorId && linha.decididaEm
          ? {
              resultado: linha.status as 'APROVADA' | 'REPROVADA',
              justificativa: linha.status === 'REPROVADA' ? linha.justificativa : null,
              decisor: { nome: autorDe(linha.decisorId, mapaUsuarios).nome },
              decididaEm: linha.decididaEm.toISOString(),
            }
          : null,
      ultimaJustificativa,
      googleAgendaUrl:
        linha.status === 'APROVADA'
          ? googleAgenda.url({
              titulo: linha.titulo,
              resumo: linha.resumo,
              inicio: linha.inicio,
              fim: linha.fim,
              horaInicio: linha.horaInicio,
              horaFim: linha.horaFim,
              periodicidade: linha.periodicidade,
              espaco: linha.espaco,
              organizadores: linha.organizadores,
            })
          : null,
    };
  }

  /** Histórico de transições, do mais recente para o mais antigo. */
  async historico(id: string): Promise<TransicaoDto[]> {
    const { leitura, usuarios } = this.deps;
    if (!(await leitura.obterLinha(id))) throw new IniciativaNaoEncontradaError();
    const linhas = await leitura.historico(id);
    const mapa = await usuarios.mapaPorIds(
      linhas.map((l) => l.por).filter((por) => por !== 'SISTEMA'),
    );
    // O banco devolve em ordem cronológica (instante e ordem de gravação); inverte para o mais recente primeiro.
    return linhas
      .map((l) => ({
        id: l.id,
        ciclo: l.ciclo,
        de: l.de,
        para: l.para,
        por: l.por === 'SISTEMA' ? 'Sistema' : (mapa.get(l.por)?.nome ?? NOME_DESCONHECIDO),
        nota: l.nota,
        justificativa: l.justificativa,
        ocorreuEm: l.ocorreuEm.toISOString(),
      }))
      .reverse();
  }

  async aprovadasParaCalendario(filtro: {
    de: string;
    ate: string;
    tipos?: TipoIniciativa[];
    autorId?: string;
  }): Promise<AprovadaParaCalendarioDto[]> {
    return this.deps.leitura.aprovadasParaCalendario(filtro);
  }

  // --- interno ---------------------------------------------------------------------------

  private async resumos(linhas: LinhaResumo[]): Promise<ResumoIniciativaDto[]> {
    if (linhas.length === 0) return [];
    const { usuarios, sugestoes } = this.deps;
    const [mapaUsuarios, mapaSugestoes] = await Promise.all([
      usuarios.mapaPorIds(linhas.map((l) => l.autorId)),
      sugestoes.contar(linhas.map((l) => ({ iniciativaId: l.id, ciclo: l.ciclo }))),
    ]);
    return linhas.map((linha) => paraResumo(linha, mapaUsuarios, mapaSugestoes));
  }

  /** Justificativa da última reprovação (também depois da ressubmissão, para a tela de edição). */
  private async ultimaJustificativa(linha: LinhaDetalhe): Promise<string | null> {
    if (linha.status === 'REPROVADA') return linha.justificativa;
    if (linha.ciclo <= 1) return null;
    const historico = await this.deps.leitura.historico(linha.id);
    const reprovacoes = historico.filter((h) => h.para === 'REPROVADA' && h.justificativa);
    reprovacoes.sort((a, b) => b.ocorreuEm.getTime() - a.ocorreuEm.getTime());
    return reprovacoes[0]?.justificativa ?? null;
  }
}

// ===== src/modules/iniciativas/application/use-cases/aplicar-prazos-vencidos.ts =====
import type { Clock } from '@/shared/domain/clock';
import { ConcorrenciaError } from '../../domain/errors/iniciativas.errors';
import type { IniciativaRepository } from '../../domain/ports/iniciativa.repository';
import type { CarregarComPrazos } from './carregar-com-prazos';

const TAMANHO_DO_LOTE = 100;

/**
 * Job de transições (RN-06): busca as iniciativas com prazo vencido, em lotes de 100, e
 * aplica as transições de cada uma. Idempotente e seguro com várias instâncias: o
 * `ConcorrenciaError` de quem perdeu a corrida é ignorado, porque outra instância já
 * processou aquela iniciativa.
 */
export class AplicarPrazosVencidos {
  constructor(
    private readonly repositorio: IniciativaRepository,
    private readonly carregar: CarregarComPrazos,
    private readonly clock: Clock,
  ) {}

  /** Devolve quantas iniciativas foram processadas. */
  async executar(): Promise<number> {
    let processadas = 0;
    const jaVistas = new Set<string>();
    for (;;) {
      const ids = (
        await this.repositorio.idsComPrazoVencido(this.clock.now(), TAMANHO_DO_LOTE)
      ).filter((id) => !jaVistas.has(id));
      if (ids.length === 0) return processadas;
      for (const id of ids) {
        jaVistas.add(id);
        try {
          await this.carregar.executar(id);
          processadas++;
        } catch (erro) {
          if (!(erro instanceof ConcorrenciaError)) throw erro;
        }
      }
    }
  }
}

// ===== src/modules/iniciativas/application/use-cases/carregar-com-prazos.ts =====
import type { Clock } from '@/shared/domain/clock';
import type { TransacaoPort } from '@/shared/domain/transacao.port';
import { ConcorrenciaError } from '../../domain/errors/iniciativas.errors';
import type { Iniciativa } from '../../domain/iniciativa';
import type { IniciativaRepository } from '../../domain/ports/iniciativa.repository';
import type { PrazosConfig } from '../../domain/ports/prazos-config';
import type { EventosPort } from '../ports/eventos.port';

const TENTATIVAS = 3;

/**
 * Carrega uma iniciativa aplicando antes as transições por prazo vencidas (RN-06). É o
 * ponto único usado por todo caso de uso que depende de status (editar, comentar, votar,
 * decidir, ler detalhe): o resultado não depende do intervalo do job.
 *
 * A aplicação roda numa transação PRÓPRIA e confirmada, antes da operação que a pediu. Assim,
 * se a operação falhar (por exemplo, editar depois do prazo → 409), a transição já ficou
 * gravada (RN-04.4). Se outra instância gravar ao mesmo tempo, relê e tenta de novo.
 */
export class CarregarComPrazos {
  constructor(
    private readonly repositorio: IniciativaRepository,
    private readonly transacao: TransacaoPort,
    private readonly eventos: EventosPort,
    private readonly clock: Clock,
    private readonly prazos: PrazosConfig,
  ) {}

  async executar(id: string): Promise<Iniciativa | null> {
    for (let tentativa = 1; ; tentativa++) {
      try {
        const iniciativa = await this.transacao.run(async () => {
          const carregada = await this.repositorio.buscarPorId(id);
          if (!carregada) return null;
          const transicoes = carregada.aplicarPrazosVencidos(this.clock, this.prazos);
          if (transicoes.length > 0) await this.repositorio.salvar(carregada);
          return carregada;
        });
        if (iniciativa) this.eventos.publicar(iniciativa.pullEvents());
        return iniciativa;
      } catch (erro) {
        if (erro instanceof ConcorrenciaError && tentativa < TENTATIVAS) continue;
        throw erro;
      }
    }
  }
}

// ===== src/modules/iniciativas/application/use-cases/tipos-iniciativa.ts =====
import type { Clock } from '@/shared/domain/clock';
import { ValidacaoError } from '@/shared/domain/domain-error';
import type { TransacaoPort } from '@/shared/domain/transacao.port';
import {
  SomenteComiteError,
  TipoEmUsoError,
  TipoNaoEncontradoError,
} from '../../domain/errors/iniciativas.errors';
import type { TipoRepository } from '../../domain/ports/tipo.repository';
import { normalizarNome, TipoDeIniciativa } from '../../domain/tipo-de-iniciativa';
import type {
  AlterarTipoInput,
  AtorTipoDto,
  CadastrarTipoInput,
  TipoIniciativaDto,
} from '../dto/tipo.dto';

const ACAO_CADASTRO = 'manter o cadastro de tipos de iniciativa';

function exigirComite(ator: AtorTipoDto): void {
  if (!ator.ehComite) throw new SomenteComiteError(ACAO_CADASTRO);
}

function paraDto(tipo: TipoDeIniciativa, contagem: Map<number, number>): TipoIniciativaDto {
  return {
    id: tipo.id,
    code: tipo.code,
    nome: tipo.nome,
    cor: tipo.cor,
    ativo: tipo.ativo,
    iniciativas: contagem.get(tipo.id) ?? 0,
  };
}

/** Não podem existir dois tipos com o mesmo nome (sem diferenciar acento e maiúsculas). */
function exigirNomeLivre(nome: string, outros: TipoDeIniciativa[]): void {
  const alvo = normalizarNome(nome);
  if (outros.some((outro) => normalizarNome(outro.nome) === alvo)) {
    throw new ValidacaoError({ nome: 'Já existe um tipo com este nome.' });
  }
}

/** Todos os tipos (ativos e inativos), com a contagem de iniciativas de cada um. */
export class ListarTipos {
  constructor(private readonly tipos: TipoRepository) {}

  async executar(filtro: { somenteAtivos?: boolean } = {}): Promise<TipoIniciativaDto[]> {
    const [todos, contagem] = await Promise.all([
      this.tipos.listar(),
      this.tipos.contarIniciativas(),
    ]);
    const lista = filtro.somenteAtivos ? todos.filter((tipo) => tipo.ativo) : todos;
    return lista.map((tipo) => paraDto(tipo, contagem));
  }
}

/** Comitê ou ADMIN cadastra um tipo (nasce ativo). Registra `CRIADO` no log. */
export class CadastrarTipo {
  constructor(
    private readonly tipos: TipoRepository,
    private readonly transacao: TransacaoPort,
    private readonly clock: Clock,
  ) {}

  async executar(ator: AtorTipoDto, entrada: CadastrarTipoInput): Promise<TipoIniciativaDto> {
    exigirComite(ator);
    const existentes = await this.tipos.listar();
    const tipo = TipoDeIniciativa.criar(entrada, new Set(existentes.map((t) => t.code)));
    exigirNomeLivre(tipo.nome, existentes);
    const id = await this.transacao.run(() =>
      this.tipos.criar(tipo, { id: ator.id, nome: ator.nome }, this.clock.now()),
    );
    return {
      id,
      code: tipo.code,
      nome: tipo.nome,
      cor: tipo.cor,
      ativo: tipo.ativo,
      iniciativas: 0,
    };
  }
}

/**
 * Comitê ou ADMIN renomeia, troca a cor e/ou ativa e desativa um tipo, mesmo com iniciativas
 * associadas (o código não muda). Registra no log cada mudança efetiva.
 */
export class AlterarTipo {
  constructor(
    private readonly tipos: TipoRepository,
    private readonly transacao: TransacaoPort,
    private readonly clock: Clock,
  ) {}

  async executar(
    id: number,
    ator: AtorTipoDto,
    entrada: AlterarTipoInput,
  ): Promise<TipoIniciativaDto> {
    exigirComite(ator);
    const tipo = await this.tipos.buscarPorId(id);
    if (!tipo) throw new TipoNaoEncontradoError();

    tipo.alterar({ nome: entrada.nome, cor: entrada.cor });
    if (entrada.nome !== undefined) {
      const outros = (await this.tipos.listar()).filter((outro) => outro.id !== tipo.id);
      exigirNomeLivre(tipo.nome, outros);
    }
    if (entrada.ativo !== undefined) tipo.definirAtivo(entrada.ativo);

    await this.transacao.run(() =>
      this.tipos.salvar(tipo, { id: ator.id, nome: ator.nome }, this.clock.now()),
    );
    const contagem = await this.tipos.contarIniciativas();
    return paraDto(tipo, contagem);
  }
}

/**
 * Comitê ou ADMIN exclui um tipo. Proibido (409) se houver iniciativas associadas, em
 * qualquer status: nesse caso, a saída é desativar. Registra `EXCLUIDO` no log.
 */
export class ExcluirTipo {
  constructor(
    private readonly tipos: TipoRepository,
    private readonly transacao: TransacaoPort,
    private readonly clock: Clock,
  ) {}

  async executar(id: number, ator: AtorTipoDto): Promise<void> {
    exigirComite(ator);
    const tipo = await this.tipos.buscarPorId(id);
    if (!tipo) throw new TipoNaoEncontradoError();
    const emUso = (await this.tipos.contarIniciativas()).get(tipo.id) ?? 0;
    if (emUso > 0) throw new TipoEmUsoError(emUso);
    await this.transacao.run(() =>
      this.tipos.excluir(
        tipo,
        tipo.registroDeExclusao(),
        { id: ator.id, nome: ator.nome },
        this.clock.now(),
      ),
    );
  }
}

export type OperacaoDeEstado =
  | 'editar'
  | 'alterarDatas'
  | 'ressubmeter'
  | 'decidir'
  | 'encerrarRefinamento'
  | 'encerrarVotacao';

/** A operação não é permitida no status atual da iniciativa (409). */
export class TransicaoInvalidaError extends DomainError {
  readonly code = 'TRANSICAO_INVALIDA';
  readonly tipo = 'conflito' as const;

  constructor(
    readonly status: StatusIniciativa,
    readonly operacao: OperacaoDeEstado,
  ) {
    super(TransicaoInvalidaError.mensagem(status, operacao));
  }

  private static mensagem(status: StatusIniciativa, operacao: OperacaoDeEstado): string {
    const rotulo = ROTULO_STATUS[status];
    switch (operacao) {
      case 'editar':
        return `Iniciativas com status "${rotulo}" não podem ser editadas.`;
      case 'ressubmeter':
        return `Iniciativas com status "${rotulo}" não podem ser ressubmetidas.`;
      case 'decidir':
        return status === 'APROVADA' || status === 'REPROVADA'
          ? 'Esta iniciativa já foi decidida.'
          : 'Esta iniciativa ainda não está em análise.';
      case 'alterarDatas':
        return 'Só iniciativas aprovadas podem ter as datas alteradas.';
      default:
        return `Operação inválida para iniciativas com status "${rotulo}".`;
    }
  }
}

/** Tipo com iniciativas associadas não pode ser excluído (só desativado). */

  readonly tipo = 'conflito' as const;

  constructor(readonly quantidade?: number) {
    super(TipoEmUsoError.mensagem(quantidade));
  }

  private static mensagem(quantidade?: number): string {

    return `Este tipo não pode ser excluído porque há ${quantidade} ${iniciativas} a ele. ${sufixo}`;
  }
}

import type { Clock } from '@/shared/domain/clock';
import type { DomainEvent } from '@/shared/domain/domain-event';
import {

  SomenteAutorError,
  SomenteComiteError,
  TransicaoInvalidaError,
} from './errors/iniciativas.errors';
import {
  DatasIniciativaAlteradas,
  IniciativaDecidida,
  IniciativaEditada,
  IniciativaRessubmetida,
  IniciativaSubmetida,
  RefinamentoEncerrado,
  VotacaoEncerrada,
} from './events/iniciativa.events';
import { MS_POR_DIA, type PrazosConfig } from './ports/prazos-config';
import type { StatusIniciativa } from './status';
import type { DadosIniciativa, DatasIniciativaBrutas } from './value-objects/dados-iniciativa';

/** Autor das transições feitas pelo job e pelos prazos. */
export const AUTOR_SISTEMA = 'SISTEMA';

export const NOTAS = {
  submetida: 'Submetida e aberta a sugestões',
  editada: 'Iniciativa editada durante o refinamento',
  refinamentoEncerrado: 'Prazo de refinamento encerrado',
  votacaoEncerrada: 'Prazo de votação encerrado',
  aprovada: 'Aprovada pelo comitê',
  reprovada: 'Reprovada pelo comitê',
  ressubmetida: (ciclo: number) => `Ressubmetida após reprovação (ciclo ${ciclo})`,
  datasAlteradas: (antes: DadosIniciativa, depois: DadosIniciativa) =>
    `Datas alteradas: de ${descreverPeriodo(antes)} para ${descreverPeriodo(depois)}`,
} as const;

/** `YYYY-MM-DD` → `DD/MM/YYYY`. */
const dataBr = (dia: string) => dia.split('-').reverse().join('/');

/** Ex.: `10/12/2026 a 11/12/2026, 09:00–18:00` ou `10/12/2026 (dia inteiro)`. */
function descreverPeriodo(dados: DadosIniciativa): string {
  const datas =
    dados.inicio === dados.fim
      ? dataBr(dados.inicio)
      : `${dataBr(dados.inicio)} a ${dataBr(dados.fim)}`;
  return dados.horaInicio
    ? `${datas}, ${dados.horaInicio}–${dados.horaFim}`
    : `${datas} (dia inteiro)`;
}

/** Uma entrada do histórico de status. */
export interface Transicao {
  de: StatusIniciativa | null;
  para: StatusIniciativa;
  /** Id do usuário ou `SISTEMA`. */
  por: string;
  nota: string;
  /** Preenchida na reprovação, para preservar o texto após a ressubmissão. */
  justificativa: string | null;
  ocorreuEm: Date;
  ciclo: number;
}

export interface IniciativaProps {
  id: string;
  autorId: string;
  dados: DadosIniciativa;
  comVotacao: boolean;
  status: StatusIniciativa;
  ciclo: number;
  fimRefinamento: Date | null;
  fimVotacao: Date | null;
  justificativa: string | null;
  decisorId: string | null;
  decididaEm: Date | null;
  version: number;
  criadaEm: Date;
}

export interface Permissoes {
  podeEditar: boolean;
  podeComentar: boolean;
  podeVotar: boolean;
  podeDecidir: boolean;
  motivoNaoPodeDecidir?: string;
  podeRessubmeter: boolean;
}

const diasEmMs = (dias: number) => Math.round(dias * MS_POR_DIA);

/**
 * Agregado `Iniciativa`: dono da máquina de estados (RN-03, 04, 06, 10, 11). O estado é
 * privado; só os métodos de comportamento o alteram e registram transições (histórico) e
 * eventos de domínio.
 */
export class Iniciativa {
  private transicoesNovas: Transicao[] = [];
  private eventos: DomainEvent[] = [];

  private constructor(
    private props: IniciativaProps,
    private ehNova: boolean,
  ) {}

  // --- criação ---------------------------------------------------------------------------

  /** RN-03: cria em `EM_ANDAMENTO`, ciclo 1, com `fimRefinamento = agora + prazo`. */
  static submeter(
    dados: DadosIniciativa,
    autorId: string,
    comVotacao: boolean,
    clock: Clock,
    prazos: PrazosConfig,
  ): Iniciativa {
    const agora = clock.now();
    const iniciativa = new Iniciativa(
      {
        id: globalThis.crypto.randomUUID(),
        autorId,
        dados,
        comVotacao,
        status: 'EM_ANDAMENTO',
        ciclo: 1,
        fimRefinamento: new Date(agora.getTime() + diasEmMs(prazos.refinamentoDias)),
        fimVotacao: null,
        justificativa: null,
        decisorId: null,
        decididaEm: null,
        version: 0,
        criadaEm: agora,
      },
      true,
    );
    iniciativa.registrar(null, 'EM_ANDAMENTO', autorId, NOTAS.submetida, agora);
    iniciativa.eventos.push(new IniciativaSubmetida(iniciativa.props.id, 1, agora));
    return iniciativa;
  }

  /** Reconstitui uma iniciativa já persistida (sem eventos nem transições pendentes). */
  static reconstituir(props: IniciativaProps): Iniciativa {
    return new Iniciativa({ ...props }, false);
  }

  // --- leitura ---------------------------------------------------------------------------

  get id(): string {
    return this.props.id;
  }
  get autorId(): string {
    return this.props.autorId;
  }
  get dados(): DadosIniciativa {
    return this.props.dados;
  }
  get comVotacao(): boolean {
    return this.props.comVotacao;
  }
  get status(): StatusIniciativa {
    return this.props.status;
  }
  get ciclo(): number {
    return this.props.ciclo;
  }
  get fimRefinamento(): Date | null {
    return this.props.fimRefinamento;
  }
  get fimVotacao(): Date | null {
    return this.props.fimVotacao;
  }
  get justificativa(): string | null {
    return this.props.justificativa;
  }
  get decisorId(): string | null {
    return this.props.decisorId;
  }
  get decididaEm(): Date | null {
    return this.props.decididaEm;
  }
  get version(): number {
    return this.props.version;
  }
  get criadaEm(): Date {
    return this.props.criadaEm;
  }
  get nova(): boolean {
    return this.ehNova;
  }

  /** Transições ainda não gravadas (o repositório as persiste junto com a iniciativa). */
  novasTransicoes(): readonly Transicao[] {
    return this.transicoesNovas;
  }

  pullEvents(): DomainEvent[] {
    const eventos = this.eventos;
    this.eventos = [];
    return eventos;
  }

  /** Chamado pelo repositório depois de gravar: limpa o pendente e avança a versão. */
  confirmarGravacao(): void {
    this.transicoesNovas = [];
    this.props = { ...this.props, version: this.props.version + 1 };
    this.ehNova = false;
  }

  // --- comportamento ---------------------------------------------------------------------

  /** RN-04: só o autor (403) e só em `EM_ANDAMENTO` (409). Pode mudar `comVotacao`. */
  editar(dados: DadosIniciativa, comVotacao: boolean, porUsuarioId: string, clock: Clock): void {
    if (porUsuarioId !== this.props.autorId) throw new SomenteAutorError('editar');
    if (this.props.status !== 'EM_ANDAMENTO') {
      throw new TransicaoInvalidaError(this.props.status, 'editar');
    }
    const agora = clock.now();
    this.props = { ...this.props, dados, comVotacao };
    this.registrar('EM_ANDAMENTO', 'EM_ANDAMENTO', porUsuarioId, NOTAS.editada, agora);
    this.eventos.push(new IniciativaEditada(this.props.id, this.props.ciclo, agora));
  }

  /**
   * RN-06: fim do refinamento. Com votação vai para `EM_VOTACAO` (com `fimVotacao =
   * fimRefinamento + prazo`); sem votação, direto para `EM_ANALISE`. O histórico registra o
   * prazo vencido, não o instante da execução.
   */
  encerrarRefinamento(prazos: PrazosConfig): void {
    if (this.props.status !== 'EM_ANDAMENTO') {
      throw new TransicaoInvalidaError(this.props.status, 'encerrarRefinamento');
    }
    const prazo = this.props.fimRefinamento ?? this.props.criadaEm;
    if (this.props.comVotacao) {
      this.props = {
        ...this.props,
        status: 'EM_VOTACAO',
        fimVotacao: new Date(prazo.getTime() + diasEmMs(prazos.votacaoDias)),
      };
      this.registrar(
        'EM_ANDAMENTO',
        'EM_VOTACAO',
        AUTOR_SISTEMA,
        NOTAS.refinamentoEncerrado,
        prazo,
      );
      this.eventos.push(
        new RefinamentoEncerrado(this.props.id, this.props.ciclo, prazo, 'EM_VOTACAO'),
      );
    } else {
      this.props = { ...this.props, status: 'EM_ANALISE' };
      this.registrar(
        'EM_ANDAMENTO',
        'EM_ANALISE',
        AUTOR_SISTEMA,
        NOTAS.refinamentoEncerrado,
        prazo,
      );
      this.eventos.push(
        new RefinamentoEncerrado(this.props.id, this.props.ciclo, prazo, 'EM_ANALISE'),
      );
    }
  }

  /** RN-06: fim da votação, vai para `EM_ANALISE`. */
  encerrarVotacao(): void {
    if (this.props.status !== 'EM_VOTACAO') {
      throw new TransicaoInvalidaError(this.props.status, 'encerrarVotacao');
    }
    const prazo = this.props.fimVotacao ?? this.props.criadaEm;
    this.props = { ...this.props, status: 'EM_ANALISE' };
    this.registrar('EM_VOTACAO', 'EM_ANALISE', AUTOR_SISTEMA, NOTAS.votacaoEncerrada, prazo);
    this.eventos.push(new VotacaoEncerrada(this.props.id, this.props.ciclo, prazo));
  }

  /**
   * RN-06: aplica as transições cujo prazo já venceu em `clock.now()`. Uma iniciativa
   * vencida há muito tempo passa pelas duas transições na mesma chamada (duas entradas).
   */
  aplicarPrazosVencidos(clock: Clock, prazos: PrazosConfig): Transicao[] {
    const antes = this.transicoesNovas.length;
    const agora = clock.now().getTime();
    if (
      this.props.status === 'EM_ANDAMENTO' &&
      this.props.fimRefinamento &&
      agora >= this.props.fimRefinamento.getTime()
    ) {
      this.encerrarRefinamento(prazos);
    }
    if (
      this.props.status === 'EM_VOTACAO' &&
      this.props.fimVotacao &&
      agora >= this.props.fimVotacao.getTime()
    ) {
      this.encerrarVotacao();
    }
    return this.transicoesNovas.slice(antes);
  }

  /**
   * RN-10: só em `EM_ANALISE`; o autor não decide a própria iniciativa (403); reprovar
   * exige justificativa (validada no caso de uso, que também a apara). Registra decisor,
   * data e o histórico.
   */
  decidir(
    resultado: 'APROVADA' | 'REPROVADA',
    justificativa: string | undefined,
    decisorId: string,
    clock: Clock,
  ): void {
    if (decisorId === this.props.autorId) throw new AutorNaoPodeDecidirError();
    if (this.props.status !== 'EM_ANALISE') {
      throw new TransicaoInvalidaError(this.props.status, 'decidir');
    }
    const agora = clock.now();
    const texto = resultado === 'REPROVADA' ? (justificativa?.trim() ?? '') : '';
    this.props = {
      ...this.props,
      status: resultado,
      decisorId,
      decididaEm: agora,
      justificativa: resultado === 'REPROVADA' ? texto : null,
    };
    this.registrar(
      'EM_ANALISE',
      resultado,
      decisorId,
      resultado === 'APROVADA' ? NOTAS.aprovada : NOTAS.reprovada,
      agora,
      resultado === 'REPROVADA' ? texto : null,
    );
    this.eventos.push(
      new IniciativaDecidida(this.props.id, this.props.ciclo, agora, resultado, decisorId),
    );
  }

  /**
   * RN-11: só o autor (403), só em `REPROVADA` (409). Incrementa o ciclo, reabre o
   * refinamento com novo prazo e limpa a decisão (a justificativa anterior fica no histórico).
   */
  ressubmeter(
    dados: DadosIniciativa,
    comVotacao: boolean,
    porUsuarioId: string,
    clock: Clock,
    prazos: PrazosConfig,
  ): void {
    if (porUsuarioId !== this.props.autorId) throw new SomenteAutorError('ressubmeter');
    if (this.props.status !== 'REPROVADA') {
      throw new TransicaoInvalidaError(this.props.status, 'ressubmeter');
    }
    const agora = clock.now();
    const ciclo = this.props.ciclo + 1;
    this.props = {
      ...this.props,
      dados,
      comVotacao,
      status: 'EM_ANDAMENTO',
      ciclo,
      fimRefinamento: new Date(agora.getTime() + diasEmMs(prazos.refinamentoDias)),
      fimVotacao: null,
      decisorId: null,
      decididaEm: null,
      justificativa: null,
    };
    this.registrar('REPROVADA', 'EM_ANDAMENTO', porUsuarioId, NOTAS.ressubmetida(ciclo), agora);
    this.eventos.push(new IniciativaRessubmetida(this.props.id, ciclo, agora));
  }

  /**
   * Muda datas e horários de uma aprovada (a série inteira das ocorrências). Só o comitê ou o
   * ADMIN (403) e só em `APROVADA` (409). O autor que é do comitê também pode: não é uma
   * decisão, então não há conflito de interesse. Registra no histórico o antes e o depois.
   */
  alterarDatas(
    datas: DatasIniciativaBrutas,
    ator: { id: string; ehComite: boolean },
    clock: Clock,
  ): void {
    if (!ator.ehComite) throw new SomenteComiteError();
    if (this.props.status !== 'APROVADA') {
      throw new TransicaoInvalidaError(this.props.status, 'alterarDatas');
    }
    const antes = this.props.dados;
    const depois = antes.comDatas(datas);
    const agora = clock.now();
    this.props = { ...this.props, dados: depois };
    this.registrar('APROVADA', 'APROVADA', ator.id, NOTAS.datasAlteradas(antes, depois), agora);
    this.eventos.push(
      new DatasIniciativaAlteradas(this.props.id, this.props.ciclo, agora, ator.id),
    );
  }

  /** O que o usuário pode fazer agora. Calculado no domínio: o front não recalcula regras. */
  permissoesPara(usuario: { id: string; ehComite: boolean }, clock: Clock): Permissoes {
    const agora = clock.now().getTime();
    const ehAutor = usuario.id === this.props.autorId;
    const refinamentoAberto =
      this.props.status === 'EM_ANDAMENTO' &&
      !(this.props.fimRefinamento && agora >= this.props.fimRefinamento.getTime());
    const votacaoAberta =
      this.props.status === 'EM_VOTACAO' &&
      !(this.props.fimVotacao && agora >= this.props.fimVotacao.getTime());
    const emAnalise = this.props.status === 'EM_ANALISE';

    const permissoes: Permissoes = {
      podeEditar: ehAutor && refinamentoAberto,
      podeComentar: refinamentoAberto,
      podeVotar: votacaoAberta,
      podeDecidir: usuario.ehComite && emAnalise && !ehAutor,
      podeRessubmeter: ehAutor && this.props.status === 'REPROVADA',
    };
    if (usuario.ehComite && emAnalise && ehAutor) {
      permissoes.motivoNaoPodeDecidir = new AutorNaoPodeDecidirError().message;
    }
    return permissoes;
  }

  // --- interno ---------------------------------------------------------------------------

  private registrar(
    de: StatusIniciativa | null,
    para: StatusIniciativa,
    por: string,
    nota: string,
    ocorreuEm: Date,
    justificativa: string | null = null,
  ): void {
    this.transicoesNovas.push({
      de,
      para,
      por,
      nota,
      justificativa,
      ocorreuEm,
      ciclo: this.props.ciclo,
    });
  }
}

import { ValidacaoError } from '@/shared/domain/domain-error';
import { FORMATO_CODIGO_TIPO } from './tipos';

export const LIMITE_NOME_TIPO = 60;
const FORMATO_COR = /^#[0-9A-Fa-f]{6}$/;

/** Campos de um tipo que aparecem no log (antes e depois). */
export interface CamposTipo {
  nome: string;
  cor: string;
  ativo: boolean;
}

export interface TipoDeIniciativaProps extends CamposTipo {
  id: number;
  code: string;
}

/** Ação registrada no log de auditoria do cadastro de tipos (lookup `iniciativas_tipos_acoes`). */
export type AcaoTipo = 'CRIADO' | 'ALTERADO' | 'ATIVADO' | 'DESATIVADO' | 'EXCLUIDO';

/** Uma entrada do log, montada pelo domínio e gravada pelo repositório na mesma transação. */
export interface RegistroTipo {
  acao: AcaoTipo;
  /** Só os campos que mudaram (`null` na criação). */
  antes: Partial<CamposTipo> | null;
  /** Só os campos que mudaram (`null` na exclusão). */
  depois: Partial<CamposTipo> | null;
}

/**
 * Nome comparável: sem acento, minúsculo e com espaços simples. "Liga  Acadêmica" e
 * "liga academica" são o mesmo nome (não podem coexistir).
 */
export function normalizarNome(nome: string): string {
  return nome
    .normalize('NFD')
    .replace(/\p{Diacritic}/gu, '')
    .trim()
    .replace(/\s+/g, ' ')
    .toLowerCase();
}

/** Código a partir do nome: "Liga Acadêmica" → `LIGA_ACADEMICA`. */
export function codigoDoNome(nome: string): string {
  const base = normalizarNome(nome)
    .toUpperCase()
    .replace(/[^A-Z0-9]+/g, '_')
    .replace(/^_+|_+$/g, '')
    .slice(0, 34);
  return base || 'TIPO';
}

function validar(campos: { nome?: unknown; cor?: unknown }): { nome: string; cor: string } {
  const erros: Record<string, string> = {};
  const nome = typeof campos.nome === 'string' ? campos.nome.trim().replace(/\s+/g, ' ') : '';
  if (!nome) erros.nome = 'Informe o nome do tipo.';
  else if (nome.length > LIMITE_NOME_TIPO)
    erros.nome = `O nome deve ter no máximo ${LIMITE_NOME_TIPO} caracteres.`;
  const cor = typeof campos.cor === 'string' ? campos.cor.trim() : '';
  if (!cor) erros.cor = 'Escolha a cor do tipo.';
  else if (!FORMATO_COR.test(cor)) erros.cor = 'Informe a cor no formato #RRGGBB.';
  if (Object.keys(erros).length > 0) throw new ValidacaoError(erros);
  return { nome, cor: cor.toUpperCase() };
}

/**
 * Tipo de iniciativa cadastrado pelo comitê ou pelo ADMIN. O `code` é gerado do nome na
 * criação e nunca muda (as iniciativas e os filtros usam o código); o nome e a cor podem ser
 * alterados. Cada mudança gera um `RegistroTipo` para o log de auditoria.
 */
export class TipoDeIniciativa {
  private registros: RegistroTipo[] = [];

  private constructor(private props: TipoDeIniciativaProps) {}

  /**
   * Cria um tipo ativo. `codigosEmUso` evita repetir o código (por exemplo, quando um tipo
   * renomeado ainda tem o código do nome antigo): acrescenta `_2`, `_3`...
   */
  static criar(
    campos: { nome?: unknown; cor?: unknown },
    codigosEmUso: ReadonlySet<string>,
  ): TipoDeIniciativa {
    const { nome, cor } = validar(campos);
    const base = codigoDoNome(nome);
    let code = base;
    for (let n = 2; codigosEmUso.has(code); n++) code = `${base}_${n}`;
    if (!FORMATO_CODIGO_TIPO.test(code)) throw new ValidacaoError({ nome: 'Nome inválido.' });
    const tipo = new TipoDeIniciativa({ id: 0, code, nome, cor, ativo: true });
    tipo.registros.push({ acao: 'CRIADO', antes: null, depois: { nome, cor, ativo: true } });
    return tipo;
  }

  static reconstituir(props: TipoDeIniciativaProps): TipoDeIniciativa {
    return new TipoDeIniciativa({ ...props });
  }

  get id(): number {
    return this.props.id;
  }
  get code(): string {
    return this.props.code;
  }
  get nome(): string {
    return this.props.nome;
  }
  get cor(): string {
    return this.props.cor;
  }
  get ativo(): boolean {
    return this.props.ativo;
  }

  /**
   * Renomeia e/ou troca a cor (campos ausentes ficam como estão). Não registra nada se nada
   * mudou de fato.
   */
  alterar(campos: { nome?: unknown; cor?: unknown }): void {
    const { nome, cor } = validar({
      nome: campos.nome === undefined ? this.props.nome : campos.nome,
      cor: campos.cor === undefined ? this.props.cor : campos.cor,
    });
    const antes: Partial<CamposTipo> = {};
    const depois: Partial<CamposTipo> = {};
    if (nome !== this.props.nome) {
      antes.nome = this.props.nome;
      depois.nome = nome;
    }
    if (cor !== this.props.cor) {
      antes.cor = this.props.cor;
      depois.cor = cor;
    }
    if (Object.keys(depois).length === 0) return;
    this.props = { ...this.props, nome, cor };
    this.registros.push({ acao: 'ALTERADO', antes, depois });
  }

  /** Ativa ou desativa. Sem efeito (nem registro) se já está no estado pedido. */
  definirAtivo(ativo: boolean): void {
    if (ativo === this.props.ativo) return;
    this.props = { ...this.props, ativo };
    this.registros.push({
      acao: ativo ? 'ATIVADO' : 'DESATIVADO',
      antes: { ativo: !ativo },
      depois: { ativo },
    });
  }

  /** Registro da exclusão. Quem confere se há iniciativas associadas é o caso de uso. */
  registroDeExclusao(): RegistroTipo {
    return {
      acao: 'EXCLUIDO',
      antes: { nome: this.props.nome, cor: this.props.cor, ativo: this.props.ativo },
      depois: null,
    };
  }

  /** Registros ainda não gravados (o repositório os grava junto com o tipo). */
  pullRegistros(): RegistroTipo[] {
    const registros = this.registros;
    this.registros = [];
    return registros;
  }
}

// ===== src/modules/iniciativas/domain/value-objects/dados-iniciativa.ts =====
import { ValidacaoError } from '@/shared/domain/domain-error';
import {
  FORMATO_CODIGO_TIPO,
  PERIODICIDADES,
  type Periodicidade,
  type TipoIniciativa,

/** Entrada bruta (como veio do HTTP). Todos os campos são validados aqui (RN-01 e RN-02). */
export interface DadosIniciativaBrutos {
  tipo?: unknown;
  titulo?: unknown;
  resumo?: unknown;
  detalhe?: unknown;
  inicio?: unknown;
  fim?: unknown;
  horaInicio?: unknown;
  horaFim?: unknown;
  periodicidade?: unknown;
  espaco?: unknown;
  organizadores?: unknown;
  membros?: unknown;
  participantes?: unknown;
}

/** Só as datas e os horários (alteração de datas de uma aprovada). */
export type DatasIniciativaBrutas = Pick<
  DadosIniciativaBrutos,
  'inicio' | 'fim' | 'horaInicio' | 'horaFim'
>;

export const LIMITES = {
  titulo: 120,
  resumo: 280,
  detalhe: 5000,
  espaco: 200,
  organizadores: 500,
  membros: 500,
  participantesMax: 100_000,
} as const;

const REGEX_DATA = /^\d{4}-\d{2}-\d{2}$/;
const REGEX_HORA = /^([01]\d|2[0-3]):[0-5]\d$/;

function dataValida(valor: string): boolean {
  if (!REGEX_DATA.test(valor)) return false;
  const [ano, mes, dia] = valor.split('-').map(Number);
  const data = new Date(Date.UTC(ano, mes - 1, dia));
  return (
    data.getUTCFullYear() === ano && data.getUTCMonth() === mes - 1 && data.getUTCDate() === dia
  );
}

/** Texto aparado, ou `''` quando não é string. */
const texto = (valor: unknown): string => (typeof valor === 'string' ? valor.trim() : '');

/**
 * Dados de uma iniciativa, já validados (imutável). `criar` acumula TODOS os erros de campo
 * num `ValidacaoError` (422 com `errors` por campo, com os nomes dos campos do DTO).
 */
export class DadosIniciativa {
  private constructor(
    readonly tipo: TipoIniciativa,
    readonly titulo: string,
    readonly resumo: string,
    readonly detalhe: string,
    /** `YYYY-MM-DD` */
    readonly inicio: string,
    readonly fim: string,
    /** `HH:mm`, ou null (dia inteiro). Sempre juntas com `horaFim`. */
    readonly horaInicio: string | null,
    readonly horaFim: string | null,
    readonly periodicidade: Periodicidade,
    readonly espaco: string,
    readonly organizadores: string,
    readonly membros: string | null,
    readonly participantes: number,
  ) {}

  /**
   * `tiposPermitidos`: códigos dos tipos que podem ser escolhidos (os ativos do cadastro, e o
   * tipo atual da iniciativa na edição). O caso de uso sempre informa; sem ele, só o formato
   * do código é conferido (usado ao revalidar dados que já foram aceitos, como em `comDatas`).
   */
  static criar(
    brutos: DadosIniciativaBrutos,
    tiposPermitidos?: ReadonlySet<string>,
  ): DadosIniciativa {
    const erros: Record<string, string> = {};

    const tipo = texto(brutos.tipo);
    if (!tipo || !FORMATO_CODIGO_TIPO.test(tipo)) erros.tipo = 'Selecione o tipo.';
    else if (tiposPermitidos && !tiposPermitidos.has(tipo))
      erros.tipo = 'Este tipo não existe ou foi desativado. Selecione outro.';

    const titulo = texto(brutos.titulo);
    if (!titulo) erros.titulo = 'Informe o título.';
    else if (titulo.length > LIMITES.titulo)
      erros.titulo = `O título deve ter no máximo ${LIMITES.titulo} caracteres.`;

    const resumo = texto(brutos.resumo);
    if (!resumo) erros.resumo = 'Escreva uma descrição resumida.';
    else if (resumo.length > LIMITES.resumo)
      erros.resumo = `A descrição resumida deve ter no máximo ${LIMITES.resumo} caracteres.`;

    const detalhe = texto(brutos.detalhe);
    if (!detalhe) erros.detalhe = 'Escreva a descrição detalhada.';
    else if (detalhe.length > LIMITES.detalhe)
      erros.detalhe = `A descrição detalhada deve ter no máximo ${LIMITES.detalhe} caracteres.`;

    const inicio = texto(brutos.inicio);
    if (!inicio) erros.inicio = 'Informe a data de início.';
    else if (!dataValida(inicio)) erros.inicio = 'Informe uma data de início válida.';

    const fim = texto(brutos.fim);
    if (!fim) erros.fim = 'Informe a data de término.';
    else if (!dataValida(fim)) erros.fim = 'Informe uma data de término válida.';
    else if (inicio && dataValida(inicio) && fim < inicio)
      erros.fim = 'A data de término deve ser igual ou posterior à de início.';

    const horaInicio = texto(brutos.horaInicio);
    const horaFim = texto(brutos.horaFim);
    if (horaInicio && !horaFim) {
      erros.horaFim = 'Informe a hora de término ou apague a de início.';
    } else if (!horaInicio && horaFim) {
      erros.horaInicio = 'Informe a hora de início ou apague a de término.';
    } else if (horaInicio && horaFim) {
      if (!REGEX_HORA.test(horaInicio)) erros.horaInicio = 'Informe uma hora de início válida.';
      if (!REGEX_HORA.test(horaFim)) erros.horaFim = 'Informe uma hora de término válida.';
      else if (REGEX_HORA.test(horaInicio) && horaFim <= horaInicio)
        erros.horaFim = 'A hora de término deve ser posterior à de início.';
    }

    const periodicidade = texto(brutos.periodicidade);
    if (!(PERIODICIDADES as readonly string[]).includes(periodicidade))
      erros.periodicidade = 'Selecione a periodicidade.';

    const espaco = texto(brutos.espaco);
    if (!espaco) erros.espaco = 'Informe o espaço utilizado.';
    else if (espaco.length > LIMITES.espaco)
      erros.espaco = `O espaço deve ter no máximo ${LIMITES.espaco} caracteres.`;

    const organizadores = texto(brutos.organizadores);
    if (!organizadores) erros.organizadores = 'Informe ao menos um organizador.';
    else if (organizadores.length > LIMITES.organizadores)
      erros.organizadores = `Os organizadores devem ter no máximo ${LIMITES.organizadores} caracteres.`;

    const membros = texto(brutos.membros);
    if (membros.length > LIMITES.membros)
      erros.membros = `Os membros devem ter no máximo ${LIMITES.membros} caracteres.`;

    const participantes = Number(brutos.participantes);
    if (!Number.isInteger(participantes) || participantes < 1)
      erros.participantes = 'Informe um número maior que zero.';
    else if (participantes > LIMITES.participantesMax)
      erros.participantes = `O número de participantes deve ser no máximo ${LIMITES.participantesMax}.`;

    if (Object.keys(erros).length > 0) {
      throw new ValidacaoError(erros);
    }

    return new DadosIniciativa(
      tipo as TipoIniciativa,
      titulo,
      resumo,
      detalhe,
      inicio,
      fim,
      horaInicio || null,
      horaFim || null,
      periodicidade as Periodicidade,
      espaco,
      organizadores,
      membros || null,
      participantes,
    );
  }

  /**
   * Cópia com novas datas e horários (os demais campos ficam como estão). Revalida com as
   * mesmas regras de `criar` (RN-02), então só os erros de data e hora podem aparecer.
   */
  comDatas(datas: DatasIniciativaBrutas): DadosIniciativa {
    return DadosIniciativa.criar({
      tipo: this.tipo,
      titulo: this.titulo,
      resumo: this.resumo,
      detalhe: this.detalhe,
      periodicidade: this.periodicidade,
      espaco: this.espaco,
      organizadores: this.organizadores,
      membros: this.membros,
      participantes: this.participantes,
      inicio: datas.inicio,
      fim: datas.fim,
      horaInicio: datas.horaInicio,
      horaFim: datas.horaFim,
    });
  }

  get temHorario(): boolean {
    return this.horaInicio !== null;
  }
}

 * Modelo de leitura das iniciativas, em SQL.
 *
 * EXCEÇÃO DOCUMENTADA À REGRA 5 (ver docs/DECISOES.md, ADR-002): esta consulta junta
 * `votacao_placar` (tabela do módulo `votacao`) para ordenar por relevância com paginação
 * no banco. A relevância (`votacao_placar.relevancia`) é mantida pelo módulo `votacao` na
 * transação do voto; aqui ela só é lida. A ordem do SQL tem de bater com `compararRelevancia`
 * do domínio de `votacao` (teste de integração RN-08.5).
 */
const COLUNAS_RESUMO = Prisma.sql`
  i.id,
  i.autor_id AS "autorId",
  t.code AS tipo,
  i.titulo,
  i.resumo,
  s.code AS status,
  i.com_votacao AS "comVotacao",
  i.ciclo,
  i.fim_refinamento AS "fimRefinamento",
  i.fim_votacao AS "fimVotacao",
  to_char(i.inicio, 'YYYY-MM-DD') AS inicio,
  to_char(i.fim, 'YYYY-MM-DD') AS fim,
  p.code AS periodicidade,
  i.espaco,
  i.criada_em AS "criadaEm",
  i.decidida_em AS "decididaEm",
  COALESCE(pl.likes, 0)::int AS likes,
  COALESCE(pl.dislikes, 0)::int AS dislikes,
  COALESCE(pl.relevancia, 0)::float8 AS relevancia`;

const JUNCOES = Prisma.sql`
  FROM iniciativas i
  JOIN iniciativas_status s ON s.id = i.status_id
  JOIN iniciativas_tipos t ON t.id = i.tipo_id
  JOIN iniciativas_periodicidades p ON p.id = i.periodicidade_id
  LEFT JOIN votacao_placar pl ON pl.iniciativa_id = i.id AND pl.ciclo = i.ciclo`;

function ordenacao(ordenar: FiltroLeitura['ordenar']): Prisma.Sql {
  switch (ordenar) {
    case 'relevancia':
      // RN-08: maior relevância; desempate por mais votos totais, criada há mais tempo e id.
      return Prisma.sql`ORDER BY COALESCE(pl.relevancia, 0) DESC,
        (COALESCE(pl.likes, 0) + COALESCE(pl.dislikes, 0)) DESC, i.criada_em ASC, i.id ASC`;
    case 'prazo':
      return Prisma.sql`ORDER BY (CASE s.code
          WHEN 'EM_ANDAMENTO' THEN i.fim_refinamento
          WHEN 'EM_VOTACAO' THEN i.fim_votacao END) ASC NULLS LAST, i.criada_em ASC, i.id ASC`;
    case 'inicio':
      return Prisma.sql`ORDER BY i.inicio ASC, i.titulo ASC, i.id ASC`;
    default:
      return Prisma.sql`ORDER BY i.criada_em DESC, i.id ASC`;
  }
}

function condicoes(filtro: Pick<FiltroLeitura, 'status' | 'tipo' | 'q' | 'autorId'>): Prisma.Sql {
  const lista: Prisma.Sql[] = [];
  if (filtro.status) lista.push(Prisma.sql`s.code = ${filtro.status}`);
  if (filtro.tipo) lista.push(Prisma.sql`t.code = ${filtro.tipo}`);
  if (filtro.autorId) lista.push(Prisma.sql`i.autor_id = ${filtro.autorId}`);
  if (filtro.q) {
    lista.push(
      Prisma.sql`strpos(lower(i.titulo || ' ' || i.resumo || ' ' || i.organizadores || ' ' || i.espaco), lower(${filtro.q})) > 0`,
    );
  }
  return lista.length > 0 ? Prisma.sql`WHERE ${Prisma.join(lista, ' AND ')}` : Prisma.empty;
}

@Injectable()
export class IniciativaListagemQuery implements IniciativaLeituraPort {
  constructor(private readonly host: PrismaTransactionHost) {}

  async listar(filtro: FiltroLeitura): Promise<{ linhas: LinhaResumo[]; total: number }> {
    const where = condicoes(filtro);
    const offset = (filtro.page - 1) * filtro.limit;
    const [linhas, total] = await Promise.all([
      this.host.client.$queryRaw<LinhaResumo[]>`
        SELECT ${COLUNAS_RESUMO} ${JUNCOES} ${where} ${ordenacao(filtro.ordenar)}
        LIMIT ${filtro.limit} OFFSET ${offset}`,
      this.host.client.$queryRaw<{ total: number }[]>`
        SELECT count(*)::int AS total ${JUNCOES} ${where}`,
    ]);
    return { linhas, total: total[0]?.total ?? 0 };
  }

  async contagemPorStatus(autorId: string): Promise<Partial<Record<StatusIniciativa, number>>> {
    const linhas = await this.host.client.$queryRaw<{ status: StatusIniciativa; total: number }[]>`
      SELECT s.code AS status, count(*)::int AS total
      FROM iniciativas i JOIN iniciativas_status s ON s.id = i.status_id
      WHERE i.autor_id = ${autorId}
      GROUP BY s.code`;
    return Object.fromEntries(linhas.map((linha) => [linha.status, linha.total]));
  }

  async contarPorStatus(status: StatusIniciativa): Promise<number> {
    const linhas = await this.host.client.$queryRaw<{ total: number }[]>`
      SELECT count(*)::int AS total
      FROM iniciativas i JOIN iniciativas_status s ON s.id = i.status_id
      WHERE s.code = ${status}`;
    return linhas[0]?.total ?? 0;
  }

  async obterLinha(id: string): Promise<LinhaDetalhe | null> {
    const linhas = await this.host.client.$queryRaw<LinhaDetalhe[]>`
      SELECT ${COLUNAS_RESUMO},
        i.detalhe,
        i.hora_inicio AS "horaInicio",
        i.hora_fim AS "horaFim",
        i.organizadores,
        i.membros,
        i.participantes,
        i.decisor_id AS "decisorId",
        i.justificativa
      ${JUNCOES}
      WHERE i.id = ${id}::uuid`;
    return linhas[0] ?? null;
  }

  async historico(id: string): Promise<LinhaHistorico[]> {
    return this.host.client.$queryRaw<LinhaHistorico[]>`
      SELECT h.id, h.ciclo, ds.code AS de, ps.code AS para, h.por, h.nota, h.justificativa,
        h.ocorreu_em AS "ocorreuEm"
      FROM iniciativas_historico h
      LEFT JOIN iniciativas_status ds ON ds.id = h.de_id
      JOIN iniciativas_status ps ON ps.id = h.para_id
      WHERE h.iniciativa_id = ${id}::uuid
      ORDER BY h.ocorreu_em ASC, h.seq ASC`;
  }

  async aprovadasParaCalendario(filtro: {
    de: string;
    ate: string;
    tipos?: TipoIniciativa[];
    autorId?: string;
  }): Promise<LinhaCalendario[]> {
    const extras: Prisma.Sql[] = [];
    if (filtro.tipos && filtro.tipos.length > 0) {
      extras.push(Prisma.sql`AND t.code IN (${Prisma.join(filtro.tipos)})`);
    }
    if (filtro.autorId) extras.push(Prisma.sql`AND i.autor_id = ${filtro.autorId}`);
    return this.host.client.$queryRaw<LinhaCalendario[]>`
      SELECT i.id, i.titulo, t.code AS tipo, i.resumo,
        to_char(i.inicio, 'YYYY-MM-DD') AS inicio,
        to_char(i.fim, 'YYYY-MM-DD') AS fim,
        i.hora_inicio AS "horaInicio", i.hora_fim AS "horaFim",
        p.code AS periodicidade, i.espaco, i.organizadores, i.autor_id AS "autorId"
      FROM iniciativas i
      JOIN iniciativas_status s ON s.id = i.status_id
      JOIN iniciativas_tipos t ON t.id = i.tipo_id
      JOIN iniciativas_periodicidades p ON p.id = i.periodicidade_id
      WHERE s.code = 'APROVADA'
        AND i.inicio < ${filtro.ate}::date AND i.fim >= ${filtro.de}::date
        ${extras.length > 0 ? Prisma.join(extras, ' ') : Prisma.empty}
      ORDER BY i.inicio ASC, i.id ASC`;
  }
}

import { Injectable } from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { PrismaTransactionHost } from '@/shared/infrastructure/prisma-transaction-host';

import { Iniciativa, type IniciativaProps } from '../../domain/iniciativa';

type LinhaIniciativa = Prisma.IniciativaGetPayload<{
  include: { tipo: true; periodicidade: true; status: true };
}>;

const dia = (data: Date): string => data.toISOString().slice(0, 10);

function paraDominio(linha: LinhaIniciativa): Iniciativa {
  const props: IniciativaProps = {
    id: linha.id,
    autorId: linha.autorId,
    dados: DadosIniciativa.criar({
      tipo: linha.tipo.code as TipoIniciativa,
      titulo: linha.titulo,
      resumo: linha.resumo,
      detalhe: linha.detalhe,
      inicio: dia(linha.inicio),
      fim: dia(linha.fim),
      horaInicio: linha.horaInicio,
      horaFim: linha.horaFim,
      periodicidade: linha.periodicidade.code as Periodicidade,
      espaco: linha.espaco,
      organizadores: linha.organizadores,
      membros: linha.membros,
      participantes: linha.participantes,
    }),
    comVotacao: linha.comVotacao,
    status: linha.status.code as StatusIniciativa,
    ciclo: linha.ciclo,
    fimRefinamento: linha.fimRefinamento,
    fimVotacao: linha.fimVotacao,
    justificativa: linha.justificativa,
    decisorId: linha.decisorId,
    decididaEm: linha.decididaEm,
    version: linha.version,
    criadaEm: linha.criadaEm,
  };
  return Iniciativa.reconstituir(props);
}

/** Campos escalares gravados em criação e atualização (as relações entram por `connect`). */
function camposDe(iniciativa: Iniciativa) {
  const { dados } = iniciativa;
  return {
    titulo: dados.titulo,
    resumo: dados.resumo,
    detalhe: dados.detalhe,
    inicio: new Date(dados.inicio),
    fim: new Date(dados.fim),
    horaInicio: dados.horaInicio,
    horaFim: dados.horaFim,
    espaco: dados.espaco,
    organizadores: dados.organizadores,
    membros: dados.membros,
    participantes: dados.participantes,
    comVotacao: iniciativa.comVotacao,
    ciclo: iniciativa.ciclo,
    fimRefinamento: iniciativa.fimRefinamento,
    fimVotacao: iniciativa.fimVotacao,
    justificativa: iniciativa.justificativa,
    decisorId: iniciativa.decisorId,
    decididaEm: iniciativa.decididaEm,
    tipo: { connect: { code: dados.tipo } },
    periodicidade: { connect: { code: dados.periodicidade } },
    status: { connect: { code: iniciativa.status } },
  };
}

@Injectable()
export class PrismaIniciativaRepository implements IniciativaRepository {
  /** Lookup de status (code → id), pequeno e imutável: carregado uma vez. */
  private statusIds?: Map<string, number>;

  constructor(private readonly host: PrismaTransactionHost) {}

  async buscarPorId(id: string): Promise<Iniciativa | null> {
    const linha = await this.host.client.iniciativa.findUnique({
      where: { id },
      include: { tipo: true, periodicidade: true, status: true },
    });
    return linha ? paraDominio(linha) : null;
  }

  async criar(iniciativa: Iniciativa): Promise<void> {
    await this.host.client.iniciativa.create({
      data: {
        id: iniciativa.id,
        autorId: iniciativa.autorId,
        criadaEm: iniciativa.criadaEm,
        version: 0,
        ...camposDe(iniciativa),
      },
    });
    await this.gravarHistorico(iniciativa);
    iniciativa.confirmarGravacao();
  }

  async salvar(iniciativa: Iniciativa): Promise<void> {
    try {
      await this.host.client.iniciativa.update({
        where: { id: iniciativa.id, version: iniciativa.version },
        data: { ...camposDe(iniciativa), version: { increment: 1 } },
      });
    } catch (erro) {
      // P2025: nenhuma linha com esse `id` e `version` (outra gravação chegou antes).
      if (erro instanceof Prisma.PrismaClientKnownRequestError && erro.code === 'P2025') {
        throw new ConcorrenciaError();
      }
      throw erro;
    }
    await this.gravarHistorico(iniciativa);
    iniciativa.confirmarGravacao();
  }

  async idsComPrazoVencido(agora: Date, limite: number): Promise<string[]> {
    const linhas = await this.host.client.iniciativa.findMany({
      where: {
        OR: [
          { status: { code: 'EM_ANDAMENTO' }, fimRefinamento: { lte: agora } },
          { status: { code: 'EM_VOTACAO' }, fimVotacao: { lte: agora } },
        ],
      },
      select: { id: true },
      orderBy: { criadaEm: 'asc' },
      take: limite,
    });
    return linhas.map((linha) => linha.id);
  }

  private async gravarHistorico(iniciativa: Iniciativa): Promise<void> {
    const transicoes = iniciativa.novasTransicoes();
    if (transicoes.length === 0) return;
    const ids = await this.idsDeStatus();
    await this.host.client.transicaoHistorico.createMany({
      data: transicoes.map((t) => ({
        iniciativaId: iniciativa.id,
        ciclo: t.ciclo,
        deId: t.de ? this.idDe(ids, t.de) : null,
        paraId: this.idDe(ids, t.para),
        por: t.por,
        nota: t.nota,
        justificativa: t.justificativa,
        ocorreuEm: t.ocorreuEm,
      })),
    });
  }

  private idDe(ids: Map<string, number>, code: string): number {
    const id = ids.get(code);
    if (id === undefined) throw new Error(`Status "${code}" não existe em iniciativas_status.`);
    return id;
  }

  private async idsDeStatus(): Promise<Map<string, number>> {
    if (!this.statusIds) {
      const linhas = await this.host.client.statusIniciativa.findMany();
      this.statusIds = new Map(linhas.map((linha) => [linha.code, linha.id]));
    }
    return this.statusIds;
  }
}

// ===== src/modules/iniciativas/infrastructure/persistence/prisma-tipo.repository.ts =====
import { Injectable } from '@nestjs/common';
import { type TipoIniciativa as LinhaTipo, Prisma } from '@prisma/client';
import { PrismaTransactionHost } from '@/shared/infrastructure/prisma-transaction-host';
import { TipoEmUsoError } from '../../domain/errors/iniciativas.errors';
import type { AutorRegistro, TipoRepository } from '../../domain/ports/tipo.repository';
import { type RegistroTipo, TipoDeIniciativa } from '../../domain/tipo-de-iniciativa';

const paraDominio = (linha: LinhaTipo): TipoDeIniciativa =>
  TipoDeIniciativa.reconstituir({
    id: linha.id,
    code: linha.code,
    nome: linha.label,
    cor: linha.cor,
    ativo: linha.ativo,
  });

const json = (valor: object | null) =>
  valor === null ? Prisma.DbNull : (valor as Prisma.InputJsonObject);

@Injectable()
export class PrismaTipoRepository implements TipoRepository {
  /** Lookup das ações do log (code → id), pequeno e imutável: carregado uma vez. */
  private acaoIds?: Map<string, number>;

  constructor(private readonly host: PrismaTransactionHost) {}

  async buscarPorId(id: number): Promise<TipoDeIniciativa | null> {
    const linha = await this.host.client.tipoIniciativa.findUnique({ where: { id } });
    return linha ? paraDominio(linha) : null;
  }

  async listar(): Promise<TipoDeIniciativa[]> {
    const linhas = await this.host.client.tipoIniciativa.findMany({
      orderBy: [{ label: 'asc' }, { id: 'asc' }],
    });
    return linhas.map(paraDominio);
  }

  async codigosAtivos(): Promise<string[]> {
    const linhas = await this.host.client.tipoIniciativa.findMany({
      where: { ativo: true },
      select: { code: true },
    });
    return linhas.map((linha) => linha.code);
  }

  async contarIniciativas(): Promise<Map<number, number>> {
    const grupos = await this.host.client.iniciativa.groupBy({
      by: ['tipoId'],
      _count: { _all: true },
    });
    return new Map(grupos.map((grupo) => [grupo.tipoId, grupo._count._all]));
  }

  async criar(tipo: TipoDeIniciativa, autor: AutorRegistro, agora: Date): Promise<number> {
    const linha = await this.host.client.tipoIniciativa.create({
      data: { code: tipo.code, label: tipo.nome, cor: tipo.cor, ativo: tipo.ativo },
    });
    await this.gravarLog(linha.id, tipo, tipo.pullRegistros(), autor, agora);
    return linha.id;
  }

  async salvar(tipo: TipoDeIniciativa, autor: AutorRegistro, agora: Date): Promise<void> {
    const registros = tipo.pullRegistros();
    if (registros.length === 0) return;
    await this.host.client.tipoIniciativa.update({
      where: { id: tipo.id },
      data: { label: tipo.nome, cor: tipo.cor, ativo: tipo.ativo },
    });
    await this.gravarLog(tipo.id, tipo, registros, autor, agora);
  }

  async excluir(
    tipo: TipoDeIniciativa,
    registro: RegistroTipo,
    autor: AutorRegistro,
    agora: Date,
  ): Promise<void> {
    // O log é gravado antes, ainda com o `tipoId`; a FK `SET NULL` o desliga na exclusão.
    await this.gravarLog(tipo.id, tipo, [registro], autor, agora);
    try {
      await this.host.client.tipoIniciativa.delete({ where: { id: tipo.id } });
    } catch (erro) {
      // P2003: a FK de `iniciativas.tipo_id` barrou (alguém usou o tipo nesse meio-tempo).
      if (erro instanceof Prisma.PrismaClientKnownRequestError && erro.code === 'P2003') {
        throw new TipoEmUsoError();
      }
      throw erro;
    }
  }

  private async gravarLog(
    tipoId: number,
    tipo: TipoDeIniciativa,
    registros: RegistroTipo[],
    autor: AutorRegistro,
    agora: Date,
  ): Promise<void> {
    if (registros.length === 0) return;
    const acoes = await this.acoes();
    await this.host.client.tipoIniciativaLog.createMany({
      data: registros.map((registro) => ({
        tipoId,
        tipoCode: tipo.code,
        tipoLabel: tipo.nome,
        acaoId: acoes.get(registro.acao) as number,
        por: autor.id,
        porNome: autor.nome,
        antes: json(registro.antes),
        depois: json(registro.depois),
        ocorreuEm: agora,
      })),
    });
  }

  private async acoes(): Promise<Map<string, number>> {
    if (!this.acaoIds) {
      const linhas = await this.host.client.acaoTipoIniciativa.findMany();
      this.acaoIds = new Map(linhas.map((linha) => [linha.code, linha.id]));
    }
    return this.acaoIds;
  }
}

// ===== src/modules/iniciativas/infrastructure/scheduler/transicoes.job.ts =====
import { Injectable, Logger } from '@nestjs/common';
import { Cron } from '@nestjs/schedule';
import { IniciativasFacade } from '../../application/iniciativas.facade';

/**
 * Job das transições por prazo (RN-06). A expressão cron e o fuso vêm do ambiente
 * (`TRANSICOES_CRON`, `APP_TIMEZONE`), lidos quando o módulo é carregado: `main.ts` carrega
 * o `.env` antes de importar a aplicação. Cada transição é idempotente e segura com várias
 * instâncias (concorrência otimista na iniciativa).
 */
@Injectable()
export class TransicoesJob {
  private readonly logger = new Logger(TransicoesJob.name);
  private emExecucao = false;

  constructor(private readonly iniciativas: IniciativasFacade) {}

  @Cron(process.env.TRANSICOES_CRON?.trim() || '*/1 * * * *', {
    name: 'iniciativas-transicoes',
    timeZone: process.env.APP_TIMEZONE?.trim() || 'America/Recife',
  })
  async executar(): Promise<void> {
    if (this.emExecucao) return;
    this.emExecucao = true;
    try {
      const processadas = await this.iniciativas.aplicarPrazosVencidos();
      if (processadas > 0)
        this.logger.log(`${processadas} iniciativa(s) com prazo vencido processada(s).`);
    } catch (erro) {
      this.logger.error(
        'Falha ao aplicar as transições por prazo.',
        erro instanceof Error ? erro.stack : erro,
      );
    } finally {
      this.emExecucao = false;
    }
  }

  /** Executa uma rodada agora e devolve quantas iniciativas processou (usado nos testes). */
  executarAgora(): Promise<number> {
    return this.iniciativas.aplicarPrazosVencidos();
  }
}

import {
  Body,
  Controller,
  Get,
  HttpCode,
  Param,
  ParseUUIDPipe,
  Patch,
  Post,
  Query,
} from '@nestjs/common';
import { ApiBearerAuth, ApiOperation, ApiResponse, ApiTags } from '@nestjs/swagger';
import { AccessRole } from '@/common/auth/access-role.enum';
import type { CurrentUser } from '@/common/auth/current-user.interface';
import { GetCurrentUser } from '@/common/auth/decorators/current-user.decorator';
import { Roles } from '@/common/auth/decorators/roles.decorator';
import type {
  AtorDto,
  DadosIniciativaInput,
  DatasIniciativaInput,
} from '../../application/dto/iniciativa.dto';
import { IniciativasFacade } from '../../application/iniciativas.facade';
import {
  DadosIniciativaDto,
  DatasIniciativaDto,
  ListagemQueryDto,
  MinhasQueryDto,
} from './iniciativas.dto';

function atorDe(user: CurrentUser): AtorDto {
  return {
    id: user.userId,
    nome: user.nome,
    ehComite: user.roles.includes(AccessRole.COMITE) || user.roles.includes(AccessRole.ADMIN),
  };
}

function pagina<T>(items: T[], total: number, page: number, limit: number) {
  return { items, meta: { page, limit, total, totalPages: Math.ceil(total / limit) } };
}

@ApiTags('Iniciativas')
@ApiBearerAuth()
@Controller({ path: 'iniciativas', version: '1' })
export class IniciativasController {
  constructor(private readonly iniciativas: IniciativasFacade) {}

  @Post()
  @ApiOperation({ summary: 'Submeter uma iniciativa (abre o refinamento)' })
  @ApiResponse({ status: 201, description: 'Iniciativa criada em Em andamento (detalhe)' })
  @ApiResponse({
    status: 422,
    description: 'Campos inválidos (RN-01 e RN-02), com `errors` por campo',
  })
  async submeter(@Body() corpo: DadosIniciativaDto, @GetCurrentUser() user: CurrentUser) {
    const id = await this.iniciativas.submeter(user.userId, corpo as DadosIniciativaInput);
    return this.iniciativas.detalhe(id, atorDe(user));
  }

  @Get()
  @ApiOperation({
    summary: 'Listar iniciativas (filtros, ordenação e paginação no servidor)',
    description:
      'Ex.: `status=EM_VOTACAO&ordenar=relevancia` (início) e `status=EM_ANDAMENTO&ordenar=prazo`.',
  })
  @ApiResponse({ status: 200, description: 'Página de resumos de iniciativas' })
  async listar(@Query() query: ListagemQueryDto) {
    const { items, total } = await this.iniciativas.listar({
      status: query.status,
      tipo: query.tipo,
      q: query.q,
      ordenar: query.ordenar,
      page: query.page,
      limit: query.limit,
    });
    return pagina(items, total, query.page, query.limit);
  }

  // Rotas estáticas ANTES de `:id`, senão `minhas` e `contagens` seriam lidas como ids.

  @Get('minhas')
  @ApiOperation({ summary: 'Minhas iniciativas, com a contagem por status' })
  @ApiResponse({ status: 200, description: 'Página de resumos e `meta.contagemPorStatus`' })
  async minhas(@Query() query: MinhasQueryDto, @GetCurrentUser() user: CurrentUser) {
    const resultado = await this.iniciativas.minhas(user.userId, {
      status: query.status,
      page: query.page,
      limit: query.limit,
    });
    const base = pagina(resultado.items, resultado.total, query.page, query.limit);
    return { ...base, meta: { ...base.meta, contagemPorStatus: resultado.contagemPorStatus } };
  }

  @Get('contagens')
  @ApiOperation({ summary: 'Contagens para o menu (iniciativas em análise)' })
  @ApiResponse({ status: 200, description: '`{ emAnalise }`' })
  contagens() {
    return this.iniciativas.contagens();
  }

  @Get(':id')
  @ApiOperation({ summary: 'Detalhe da iniciativa, com permissões calculadas para o usuário' })
  @ApiResponse({ status: 200, description: 'Detalhe (inclui `permissoes`, `meuVoto` e `decisao`)' })
  @ApiResponse({ status: 404, description: 'Iniciativa não encontrada' })
  detalhe(@Param('id', ParseUUIDPipe) id: string, @GetCurrentUser() user: CurrentUser) {
    return this.iniciativas.detalhe(id, atorDe(user));
  }

  @Patch(':id')
  @ApiOperation({ summary: 'Editar a própria iniciativa (só em Em andamento)' })
  @ApiResponse({ status: 200, description: 'Detalhe atualizado' })
  @ApiResponse({ status: 403, description: 'Só o autor edita' })
  @ApiResponse({ status: 409, description: 'Status não permite edição' })
  @ApiResponse({ status: 422, description: 'Campos inválidos' })
  async editar(
    @Param('id', ParseUUIDPipe) id: string,
    @Body() corpo: DadosIniciativaDto,
    @GetCurrentUser() user: CurrentUser,
  ) {
    await this.iniciativas.editar(id, user.userId, corpo as DadosIniciativaInput);
    return this.iniciativas.detalhe(id, atorDe(user));
  }

  @Patch(':id/datas')
  @Roles(AccessRole.COMITE, AccessRole.ADMIN)
  @ApiOperation({
    summary: 'Alterar datas e horários de uma aprovada (comitê ou ADMIN)',
    description:
      'Muda início, término e horários da iniciativa; todas as ocorrências do calendário são recalculadas. A periodicidade não muda.',
  })
  @ApiResponse({ status: 200, description: 'Detalhe atualizado' })
  @ApiResponse({ status: 403, description: 'Só o comitê ou o ADMIN' })
  @ApiResponse({ status: 404, description: 'Iniciativa não encontrada' })
  @ApiResponse({ status: 409, description: 'Só iniciativas aprovadas' })
  @ApiResponse({ status: 422, description: 'Datas ou horários inválidos, com `errors` por campo' })
  async alterarDatas(
    @Param('id', ParseUUIDPipe) id: string,
    @Body() corpo: DatasIniciativaDto,
    @GetCurrentUser() user: CurrentUser,
  ) {
    const ator = atorDe(user);
    await this.iniciativas.alterarDatas(id, ator, corpo as DatasIniciativaInput);
    return this.iniciativas.detalhe(id, ator);
  }

  @Post(':id/ressubmissao')
  @HttpCode(200)
  @ApiOperation({ summary: 'Ressubmeter uma iniciativa reprovada (novo ciclo)' })
  @ApiResponse({ status: 200, description: 'Detalhe no ciclo seguinte, em Em andamento' })
  @ApiResponse({ status: 403, description: 'Só o autor ressubmete' })
  @ApiResponse({ status: 409, description: 'Só iniciativas reprovadas' })
  @ApiResponse({ status: 422, description: 'Campos inválidos' })
  async ressubmeter(
    @Param('id', ParseUUIDPipe) id: string,
    @Body() corpo: DadosIniciativaDto,
    @GetCurrentUser() user: CurrentUser,
  ) {
    await this.iniciativas.ressubmeter(id, user.userId, corpo as DadosIniciativaInput);
    return this.iniciativas.detalhe(id, atorDe(user));
  }

  @Get(':id/historico')
  @ApiOperation({ summary: 'Histórico de transições de status (mais recente primeiro)' })
  @ApiResponse({ status: 200, description: 'Lista de transições' })
  @ApiResponse({ status: 404, description: 'Iniciativa não encontrada' })
  historico(@Param('id', ParseUUIDPipe) id: string) {
    return this.iniciativas.historico(id);
  }
}

import { createZodDto } from 'nestjs-zod';
import { z } from 'zod';
import { STATUS_INICIATIVA } from '../../domain/status';
import { FORMATO_CODIGO_TIPO, PERIODICIDADES } from '../../domain/tipos';

/**
 * Corpo de `POST /iniciativas`, `PATCH /iniciativas/:id` e `POST /iniciativas/:id/ressubmissao`.
 *
 * O schema só garante os TIPOS; as regras de RN-01 e RN-02 (obrigatoriedade, limites, datas e
 * horários) vivem no domínio (`DadosIniciativa`), que acumula as mensagens por campo e as
 * devolve num 422 com `errors`. Por isso os campos são opcionais aqui.
 */
export const DadosIniciativaSchema = z.object({
  tipo: z
    .string()
    .optional()
    .describe('Código de um tipo ativo do cadastro (GET /tipos-iniciativa), por exemplo PALESTRA'),
  titulo: z.string().optional().describe('Título (até 120 caracteres)'),
  resumo: z.string().optional().describe('Descrição resumida (até 280 caracteres)'),
  detalhe: z.string().optional().describe('Descrição detalhada (até 5000 caracteres)'),
  inicio: z.string().optional().describe('Data de início (YYYY-MM-DD)'),
  fim: z.string().optional().describe('Data de término (YYYY-MM-DD), igual ou posterior ao início'),
  horaInicio: z
    .string()
    .nullable()
    .optional()
    .describe('Hora de início (HH:mm), junto com horaFim'),
  horaFim: z
    .string()
    .nullable()
    .optional()
    .describe('Hora de término (HH:mm), posterior ao início'),
  periodicidade: z
    .string()
    .optional()
    .describe(`Periodicidade: ${PERIODICIDADES.join(', ')}`),
  espaco: z.string().optional().describe('Espaço utilizado (até 200 caracteres)'),
  organizadores: z.string().optional().describe('Organizadores, separados por vírgula (até 500)'),
  membros: z.string().nullable().optional().describe('Membros (opcional, até 500)'),
  participantes: z.number().optional().describe('Quantidade de participantes (1 a 100000)'),
  comVotacao: z.boolean().optional().describe('Enviar para votação geral (padrão: true)'),
});

export class DadosIniciativaDto extends createZodDto(DadosIniciativaSchema) {}

/**
 * Corpo de `PATCH /iniciativas/:id/datas`. Como em `DadosIniciativaSchema`, só os tipos;
 * as regras de data e hora (RN-02) ficam no domínio e voltam num 422 com `errors`.
 */
export const DatasIniciativaSchema = z.object({
  inicio: z.string().optional().describe('Data de início (YYYY-MM-DD)'),
  fim: z.string().optional().describe('Data de término (YYYY-MM-DD), igual ou posterior ao início'),
  horaInicio: z
    .string()
    .nullable()
    .optional()
    .describe('Hora de início (HH:mm), junto com horaFim; null para dia inteiro'),
  horaFim: z
    .string()
    .nullable()
    .optional()
    .describe('Hora de término (HH:mm), posterior ao início; null para dia inteiro'),
});

export class DatasIniciativaDto extends createZodDto(DatasIniciativaSchema) {}

  alterar(
    @Param('id', ParseIntPipe) id: number,
    @Body() corpo: AlterarTipoDto,
    @GetCurrentUser() user: CurrentUser,
  ) {
    return this.alterarTipo.executar(id, atorDe(user), corpo as AlterarTipoInput);
  }

  @Delete(':id')
  @HttpCode(204)
  @Roles(AccessRole.COMITE, AccessRole.ADMIN)
  @ApiOperation({ summary: 'Excluir um tipo sem iniciativas associadas (comitê ou ADMIN)' })
  @ApiResponse({ status: 204, description: 'Excluído' })
  @ApiResponse({ status: 403, description: 'Só o comitê ou o ADMIN' })
  @ApiResponse({ status: 404, description: 'Tipo não encontrado' })
  @ApiResponse({ status: 409, description: 'Há iniciativas associadas ao tipo' })
  async excluir(@Param('id', ParseIntPipe) id: number, @GetCurrentUser() user: CurrentUser) {
    await this.excluirTipo.executar(id, atorDe(user));
  }
}

import { createZodDto } from 'nestjs-zod';
import { z } from 'zod';

/**
 * Corpos do cadastro de tipos. Como nos outros DTOs, o schema só garante os TIPOS; as regras
 * (obrigatório, tamanho, formato da cor, nome repetido) ficam no domínio e voltam num 422
 * com `errors` por campo.
 */
export const CadastrarTipoSchema = z.object({
  nome: z.string().optional().describe('Nome do tipo (até 60 caracteres), por exemplo "Palestra"'),
  cor: z.string().optional().describe('Cor no formato #RRGGBB'),
});

export class CadastrarTipoDto extends createZodDto(CadastrarTipoSchema) {}

export const AlterarTipoSchema = z.object({
  nome: z.string().optional().describe('Novo nome (o código do tipo não muda)'),
  cor: z.string().optional().describe('Nova cor, #RRGGBB'),
  ativo: z.boolean().optional().describe('false desativa: some da seleção de novas iniciativas'),
});

export class AlterarTipoDto extends createZodDto(AlterarTipoSchema) {}

export const ListarTiposQuerySchema = z.object({
  ativos: z
    .enum(['true', 'false'])
    .optional()
    .transform((valor) => valor === 'true')
    .describe('true: só os tipos ativos (os que podem ser escolhidos)'),
});

export class ListarTiposQueryDto extends createZodDto(ListarTiposQuerySchema) {}
