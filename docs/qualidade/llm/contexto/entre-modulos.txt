// ===== src/shared/domain/domain-error.ts =====
/**
 * Categoria de um erro de domínio. É semântica de negócio, não HTTP: a camada de
 * apresentação (`HttpExceptionFilter`) traduz cada categoria para um status.
 *
 *  - `validacao`: dado inválido (422, com `errors` por campo quando é um `ValidacaoError`);
 *  - `proibido`: o usuário não pode fazer isso (403);
 *  - `nao_encontrado`: o recurso não existe (404);
 *  - `conflito`: o estado atual impede a operação, ou houve concorrência (409).
 */
export type TipoErroDominio = 'validacao' | 'proibido' | 'nao_encontrado' | 'conflito';

/**
 * Base dos erros de domínio. Cada erro carrega um `code` estável, uma categoria e uma
 * mensagem em português pronta para a tela.
 */
export abstract class DomainError extends Error {
  abstract readonly code: string;
  abstract readonly tipo: TipoErroDominio;

  constructor(message: string) {
    super(message);
    this.name = new.target.name;
  }
}

/** Erro de validação com a mensagem de cada campo inválido (nomes dos campos do DTO). */
export class ValidacaoError extends DomainError {
  readonly code = 'VALIDACAO';
  readonly tipo = 'validacao' as const;

  constructor(
    readonly campos: Record<string, string>,
    mensagem = 'Um ou mais campos precisam ser corrigidos.',
  ) {
    super(mensagem);
  }
}

import type { Clock } from '@/shared/domain/clock';
import type { UsuarioRepository } from '../domain/ports/usuario.repository';
import type { RegistrarUsuarioInput, UsuarioDto } from './dto/usuario.dto';
import { ObterUsuario, ObterUsuarios } from './queries/obter-usuarios';
import { RegistrarUsuario } from './use-cases/registrar-usuario';

export type { RegistrarUsuarioInput, UsuarioDto };

// Limite do cache de sincronização: evita crescer sem controle em instâncias longevas.
const LIMITE_CACHE = 5000;
// Enquanto nome e perfil não mudam, o usuário só é regravado a cada 5 minutos.
const TTL_CACHE_MS = 5 * 60 * 1000;

interface Sincronizacao {
  assinatura: string;
  ate: number;
}

/**
 * API pública do módulo `identidade`. Os outros módulos só falam com `identidade`
 * por aqui, e sempre por portas implementadas em `infrastructure/adapters`.
 */
export class IdentidadeFacade {
  private readonly registrarUsuario: RegistrarUsuario;
  private readonly obterUsuario: ObterUsuario;
  private readonly obterUsuarios: ObterUsuarios;
  /** id → última gravação (assinatura "nome|perfil" e validade), para não escrever a cada requisição. */
  private readonly sincronizados = new Map<string, Sincronizacao>();

  constructor(
    usuarios: UsuarioRepository,
    private readonly clock: Clock,
  ) {
    this.registrarUsuario = new RegistrarUsuario(usuarios);
    this.obterUsuario = new ObterUsuario(usuarios);
    this.obterUsuarios = new ObterUsuarios(usuarios);
  }

  /**
   * Garante que o usuário do token existe localmente com nome e perfil atuais. Grava
   * quando algo mudou desde o último registro desta instância ou quando o cache de 5
   * minutos expirou.
   */
  async garantirUsuario(input: RegistrarUsuarioInput): Promise<void> {
    const assinatura = `${input.nome}|${input.perfil}`;
    const agora = this.clock.now().getTime();
    const anterior = this.sincronizados.get(input.id);
    if (anterior && anterior.assinatura === assinatura && anterior.ate > agora) {
      return;
    }
    await this.registrarUsuario.executar(input);
    if (this.sincronizados.size >= LIMITE_CACHE) {
      this.sincronizados.clear();
    }
    this.sincronizados.set(input.id, { assinatura, ate: agora + TTL_CACHE_MS });
  }

  obterPorId(id: string): Promise<UsuarioDto> {
    return this.obterUsuario.executar(id);
  }

  usuariosPorIds(ids: string[]): Promise<UsuarioDto[]> {
    return this.obterUsuarios.executar(ids);
  }

  /** Mapa id → usuário, para exibir autores, comentaristas e decisores. */
  async mapaPorIds(ids: string[]): Promise<Map<string, UsuarioDto>> {
    const usuarios = await this.obterUsuarios.executar(ids);
    return new Map(usuarios.map((usuario) => [usuario.id, usuario]));
  }
}

export interface RegistrarDecisaoInput {
  iniciativaId: string;
  resultado: 'APROVADA' | 'REPROVADA';
  justificativa?: string | null;
  /** Quem decide. O autor da iniciativa nunca pode decidir sobre ela (RN-10). */
  decisorId: string;
}

export interface FiltroCalendarioInput {
  /** Início do intervalo, `YYYY-MM-DD` (inclusivo). */
  de: string;
  /** Fim do intervalo, `YYYY-MM-DD` (exclusivo). */
  ate: string;
  tipos?: TipoIniciativa[];
  /** Só as iniciativas deste autor. */
  autorId?: string;
}

/**
 * API pública do módulo `iniciativas`. Os outros módulos só falam com ele por aqui, e
 * sempre por portas implementadas em `infrastructure/adapters` do módulo consumidor.
 * É o único módulo que muda o status de uma iniciativa.
 */
export class IniciativasFacade {
  constructor(
    private readonly submeterIniciativa: SubmeterIniciativa,
    private readonly editarIniciativa: EditarIniciativa,
    private readonly ressubmeterIniciativa: RessubmeterIniciativa,
    private readonly decidirIniciativa: DecidirIniciativa,
    private readonly consultas: ConsultasIniciativas,
    private readonly carregar: CarregarComPrazos,
    private readonly job: AplicarPrazosVencidos,
    private readonly alterarDatasIniciativa: AlterarDatasIniciativa,
  ) {}

  // --- contrato usado pelos outros módulos --------------------------------------------------

  /**
   * Estado da iniciativa para validar operações de outros módulos (votar, comentar).
   * Aplica antes as transições por prazo vencidas (RN-06). `null` se não existir.
   */
  async obterEstado(iniciativaId: string): Promise<EstadoIniciativaDto | null> {
    const iniciativa = await this.carregar.executar(iniciativaId);
    if (!iniciativa) return null;
    return {
      id: iniciativa.id,
      autorId: iniciativa.autorId,
      status: iniciativa.status,
      ciclo: iniciativa.ciclo,
      comVotacao: iniciativa.comVotacao,
      fimRefinamento: iniciativa.fimRefinamento?.toISOString() ?? null,
      fimVotacao: iniciativa.fimVotacao?.toISOString() ?? null,
    };
  }

  /**
   * Registra a decisão do comitê (RN-10). Lança erros de domínio: `proibido` se o decisor
   * é o autor, `conflito` fora de Em análise ou se já foi decidida, `nao_encontrado` se
   * não existir. A validação da justificativa fica no caso de uso de `avaliacao`.
   */
  registrarDecisao(input: RegistrarDecisaoInput): Promise<void> {
    return this.decidirIniciativa.executar(
      input.iniciativaId,
      input.decisorId,
      input.resultado,
      input.justificativa,
    );
  }

  /** Listagem paginada de resumos, com filtros e ordenação no banco (RNF-05). */
  listar(filtro: FiltroListagemDto): Promise<PaginaDto<ResumoIniciativaDto>> {
    return this.consultas.listar(filtro);
  }

  /** Detalhe com permissões calculadas para o ator. Lança `nao_encontrado`. */
  detalhe(iniciativaId: string, ator: AtorDto): Promise<DetalheIniciativaDto> {
    return this.consultas.detalhe(iniciativaId, ator);
  }

  /** Iniciativas aprovadas cujo período pode cruzar o intervalo, para gerar ocorrências. */
  aprovadasParaCalendario(filtro: FiltroCalendarioInput): Promise<AprovadaParaCalendarioDto[]> {
    return this.consultas.aprovadasParaCalendario(filtro);
  }

  // --- usado pelos controllers e pelo job do próprio módulo ---------------------------------

  submeter(autorId: string, entrada: DadosIniciativaInput): Promise<string> {
    return this.submeterIniciativa.executar(autorId, entrada);
  }

  editar(id: string, usuarioId: string, entrada: DadosIniciativaInput): Promise<void> {
    return this.editarIniciativa.executar(id, usuarioId, entrada);
  }

  ressubmeter(id: string, usuarioId: string, entrada: DadosIniciativaInput): Promise<void> {
    return this.ressubmeterIniciativa.executar(id, usuarioId, entrada);
  }

  /** Comitê ou ADMIN muda datas e horários de uma aprovada (a série inteira). */
  alterarDatas(id: string, ator: AtorDto, datas: DatasIniciativaInput): Promise<void> {
    return this.alterarDatasIniciativa.executar(id, ator, datas);
  }

  minhas(autorId: string, filtro: { status?: StatusIniciativa; page: number; limit: number }) {
    return this.consultas.minhas(autorId, filtro);
  }

  contagens(): Promise<{ emAnalise: number }> {
    return this.consultas.contagens();
  }

  historico(iniciativaId: string): Promise<TransicaoDto[]> {
    return this.consultas.historico(iniciativaId);
  }

  /** Aplica as transições por prazo vencidas (job). Devolve quantas iniciativas processou. */
  aplicarPrazosVencidos(): Promise<number> {
    return this.job.executar();
  }
}

import type { TransacaoPort } from '@/shared/domain/transacao.port';
import type {
  ParIniciativaCiclo,
  Placar,
  VotacaoRepository,
} from '../domain/ports/votacao.repository';
import type { MeuVoto } from '../domain/voto';
import type { IniciativaPort } from './ports/iniciativa.port';
import {
  RegistrarVoto,
  type RegistrarVotoInput,
  type ResultadoVotoDto,
} from './use-cases/registrar-voto';

export type { MeuVoto, ParIniciativaCiclo, Placar, RegistrarVotoInput, ResultadoVotoDto };

const SEM_VOTOS: Placar = { likes: 0, dislikes: 0, relevancia: 0 };

/** API pública do módulo `votacao`. */
export class VotacaoFacade {
  private readonly registrarVoto: RegistrarVoto;

  constructor(
    private readonly votacao: VotacaoRepository,
    iniciativas: IniciativaPort,
    transacao: TransacaoPort,
  ) {
    this.registrarVoto = new RegistrarVoto(votacao, iniciativas, transacao);
  }

  /** RN-07. Erros: `nao_encontrado`, `conflito` (fora de Em votação), `validacao`. */
  votar(input: RegistrarVotoInput): Promise<ResultadoVotoDto> {
    return this.registrarVoto.executar(input);
  }

  /** Placar do ciclo. Zeros se não houver linha. */
  async placar(iniciativaId: string, ciclo: number): Promise<Placar> {
    return (await this.votacao.buscarPlacar(iniciativaId, ciclo)) ?? { ...SEM_VOTOS };
  }

  /** Placares em lote. Chave = `iniciativaId`; ausente = sem votos (zeros). */
  placares(pares: ParIniciativaCiclo[]): Promise<Map<string, Placar>> {
    if (pares.length === 0) return Promise.resolve(new Map());
    return this.votacao.buscarPlacares(pares);
  }

  /** Voto do usuário no ciclo: 1, -1 ou 0. */
  meuVoto(iniciativaId: string, ciclo: number, usuarioId: string): Promise<MeuVoto> {
    return this.votacao.buscarVoto(iniciativaId, ciclo, usuarioId);
  }
}
