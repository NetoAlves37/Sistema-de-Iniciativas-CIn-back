import type { TransacaoPort } from '@/shared/domain/transacao.port';
import {
  IniciativaNaoEncontradaError,
  VotacaoEncerradaError,
} from '../../domain/errors/votacao.errors';
import type { VotacaoRepository } from '../../domain/ports/votacao.repository';
import { wilson } from '../../domain/relevancia';
import { type AcaoVoto, decidirVoto, type MeuVoto, validarValorVoto } from '../../domain/voto';
import type { IniciativaPort } from '../ports/iniciativa.port';

export interface RegistrarVotoInput {
  iniciativaId: string;
  usuarioId: string;
  valor: number;
}

export interface ResultadoVotoDto {
  meuVoto: MeuVoto;
  likes: number;
  dislikes: number;
  relevancia: number;
  acao: AcaoVoto;
}

const STATUS_VOTACAO = 'EM_VOTACAO';

/** RN-07 e RN-08. O autor também pode votar. Tudo é por `(iniciativa, ciclo)`. */
export class RegistrarVoto {
  constructor(
    private readonly votacao: VotacaoRepository,
    private readonly iniciativas: IniciativaPort,
    private readonly transacao: TransacaoPort,
  ) {}

  async executar(input: RegistrarVotoInput): Promise<ResultadoVotoDto> {
    const valor = validarValorVoto(input.valor);
    const estado = await this.iniciativas.obterEstado(input.iniciativaId);
    if (!estado) throw new IniciativaNaoEncontradaError();
    if (estado.status !== STATUS_VOTACAO) throw new VotacaoEncerradaError();

    const { id: iniciativaId, ciclo } = estado;
    return this.transacao.run(async () => {
      await this.votacao.garantirPlacar(iniciativaId, ciclo);
      await this.votacao.travarPlacar(iniciativaId, ciclo);

      const atual = await this.votacao.buscarVoto(iniciativaId, ciclo, input.usuarioId);
      const decisao = decidirVoto(atual, valor);
      if (decisao.meuVoto === 0) {
        await this.votacao.removerVoto(iniciativaId, ciclo, input.usuarioId);
      } else {
        await this.votacao.gravarVoto(iniciativaId, ciclo, input.usuarioId, decisao.meuVoto);
      }

      const { likes, dislikes } = await this.votacao.contarVotos(iniciativaId, ciclo);
      const relevancia = wilson(likes, likes + dislikes);
      await this.votacao.salvarPlacar(iniciativaId, ciclo, { likes, dislikes, relevancia });

      return { meuVoto: decisao.meuVoto, likes, dislikes, relevancia, acao: decisao.acao };
    });
  }
}

} from './use-cases/registrar-voto';

const SEM_VOTOS: Placar = { likes: 0, dislikes: 0, relevancia: 0 };

/** API pública do módulo `votacao`. */
export class VotacaoFacade {
  private readonly registrarVoto: RegistrarVoto;

  constructor(
    private readonly votacao: VotacaoRepository,
    iniciativas: IniciativaPort,
    transacao: TransacaoPort,
  ) {
    this.registrarVoto = new RegistrarVoto(votacao, iniciativas, transacao);
  }

  /** RN-07. Erros: `nao_encontrado`, `conflito` (fora de Em votação), `validacao`. */
  votar(input: RegistrarVotoInput): Promise<ResultadoVotoDto> {
    return this.registrarVoto.executar(input);
  }

  /** Placar do ciclo. Zeros se não houver linha. */
  async placar(iniciativaId: string, ciclo: number): Promise<Placar> {
    return (await this.votacao.buscarPlacar(iniciativaId, ciclo)) ?? { ...SEM_VOTOS };
  }

  /** Placares em lote. Chave = `iniciativaId`; ausente = sem votos (zeros). */
  placares(pares: ParIniciativaCiclo[]): Promise<Map<string, Placar>> {
    if (pares.length === 0) return Promise.resolve(new Map());
    return this.votacao.buscarPlacares(pares);
  }

  /** Voto do usuário no ciclo: 1, -1 ou 0. */
  meuVoto(iniciativaId: string, ciclo: number, usuarioId: string): Promise<MeuVoto> {
    return this.votacao.buscarVoto(iniciativaId, ciclo, usuarioId);
  }
}

  contarVotos(iniciativaId: string, ciclo: number): Promise<{ likes: number; dislikes: number }>;

// ===== src/modules/votacao/domain/relevancia.ts =====
/** Quantil de 97,5% da normal padrão (intervalo de confiança de 95%). */
const Z = 1.96;

/**
 * Limite inferior do intervalo de Wilson para a proporção de votos positivos (RN-08).
 * Sem votos, a relevância é 0.
 */
export function wilson(positivos: number, total: number): number {
  if (total === 0) return 0;
  const p = positivos / total;
  const z2 = Z * Z;
  const centro = p + z2 / (2 * total);
  const margem = Z * Math.sqrt((p * (1 - p) + z2 / (4 * total)) / total);
  return (centro - margem) / (1 + z2 / total);
}

export interface ItemParaRelevancia {
  id: string;
  likes: number;
  dislikes: number;
  criadaEm: Date;
  /** Relevância já calculada (coluna do placar). Se ausente, é calculada de likes/dislikes. */
  relevancia?: number;
}

export const relevanciaDe = (
  item: Pick<ItemParaRelevancia, 'likes' | 'dislikes' | 'relevancia'>,
): number => item.relevancia ?? wilson(item.likes, item.likes + item.dislikes);

/**
 * Comparador para `Array.sort`: negativo se `a` vem antes de `b`. Maior relevância; depois
 * mais votos totais; depois `criadaEm` mais antiga; depois `id` (ordem determinística).
 */
export function compararRelevancia(a: ItemParaRelevancia, b: ItemParaRelevancia): number {
  const porRelevancia = relevanciaDe(b) - relevanciaDe(a);
  if (porRelevancia !== 0) return porRelevancia;
  const porVotos = b.likes + b.dislikes - (a.likes + a.dislikes);
  if (porVotos !== 0) return porVotos;
  const porData = a.criadaEm.getTime() - b.criadaEm.getTime();
  if (porData !== 0) return porData;
  if (a.id === b.id) return 0;
  return a.id < b.id ? -1 : 1;
}

// ===== src/modules/votacao/domain/voto.ts =====
import { ValidacaoError } from '@/shared/domain/domain-error';

export type ValorVoto = 1 | -1;
export type MeuVoto = 1 | -1 | 0;
export type AcaoVoto = 'REGISTRADO' | 'ALTERADO' | 'RETIRADO';

export const LIKE: ValorVoto = 1;
export const DISLIKE: ValorVoto = -1;

export function ehValorVoto(valor: unknown): valor is ValorVoto {
  return valor === LIKE || valor === DISLIKE;
}

/** Garante que `valor` é +1 ou −1; senão lança `ValidacaoError({ valor })`. */
export function validarValorVoto(valor: unknown): ValorVoto {
  if (!ehValorVoto(valor)) {
    throw new ValidacaoError({ valor: 'O voto deve ser 1 (curtir) ou -1 (não curtir).' });
  }
  return valor;
}

export interface DecisaoDeVoto {
  /** Voto do usuário depois da operação (0 = sem voto). */
  meuVoto: MeuVoto;
  acao: AcaoVoto;
}

/**
 * RN-07: o mesmo valor que o usuário já tem retira o voto; o valor oposto troca; sem voto
 * anterior, registra.
 */
export function decidirVoto(atual: MeuVoto, novo: ValorVoto): DecisaoDeVoto {
  if (atual === 0) return { meuVoto: novo, acao: 'REGISTRADO' };
  if (atual === novo) return { meuVoto: 0, acao: 'RETIRADO' };
  return { meuVoto: novo, acao: 'ALTERADO' };
}

// ===== src/modules/votacao/infrastructure/persistence/prisma-votacao.repository.ts =====
import { Injectable } from '@nestjs/common';
import { PrismaTransactionHost } from '@/shared/infrastructure/prisma-transaction-host';
import type {
  ParIniciativaCiclo,
  Placar,
  VotacaoRepository,
} from '../../domain/ports/votacao.repository';
import type { MeuVoto, ValorVoto } from '../../domain/voto';

@Injectable()
export class PrismaVotacaoRepository implements VotacaoRepository {
  constructor(private readonly host: PrismaTransactionHost) {}

  async garantirPlacar(iniciativaId: string, ciclo: number): Promise<void> {
    await this.host.client.$executeRaw`
      INSERT INTO votacao_placar (iniciativa_id, ciclo, likes, dislikes, relevancia)
      VALUES (${iniciativaId}::uuid, ${ciclo}, 0, 0, 0)
      ON CONFLICT (iniciativa_id, ciclo) DO NOTHING`;
  }

  async travarPlacar(iniciativaId: string, ciclo: number): Promise<void> {
    await this.host.client.$queryRaw`
      SELECT iniciativa_id FROM votacao_placar
      WHERE iniciativa_id = ${iniciativaId}::uuid AND ciclo = ${ciclo}
      FOR UPDATE`;
  }

  async buscarVoto(iniciativaId: string, ciclo: number, usuarioId: string): Promise<MeuVoto> {
    const voto = await this.host.client.voto.findUnique({
      where: { iniciativaId_ciclo_usuarioId: { iniciativaId, ciclo, usuarioId } },
    });
    return voto?.valor === 1 ? 1 : voto?.valor === -1 ? -1 : 0;
  }

  async gravarVoto(
    iniciativaId: string,
    ciclo: number,
    usuarioId: string,
    valor: ValorVoto,
  ): Promise<void> {
    await this.host.client.voto.upsert({
      where: { iniciativaId_ciclo_usuarioId: { iniciativaId, ciclo, usuarioId } },
      create: { iniciativaId, ciclo, usuarioId, valor },
      update: { valor },
    });
  }

  async removerVoto(iniciativaId: string, ciclo: number, usuarioId: string): Promise<void> {
    await this.host.client.voto.deleteMany({ where: { iniciativaId, ciclo, usuarioId } });
  }

  async contarVotos(
    iniciativaId: string,
    ciclo: number,
  ): Promise<{ likes: number; dislikes: number }> {
    const client = this.host.client;
    const [likes, dislikes] = await Promise.all([
      client.voto.count({ where: { iniciativaId, ciclo, valor: 1 } }),
      client.voto.count({ where: { iniciativaId, ciclo, valor: -1 } }),
    ]);
    return { likes, dislikes };
  }

  async salvarPlacar(iniciativaId: string, ciclo: number, placar: Placar): Promise<void> {
    await this.host.client.placar.update({
      where: { iniciativaId_ciclo: { iniciativaId, ciclo } },
      data: { likes: placar.likes, dislikes: placar.dislikes, relevancia: placar.relevancia },
    });
  }

  async buscarPlacar(iniciativaId: string, ciclo: number): Promise<Placar | null> {
    const linha = await this.host.client.placar.findUnique({
      where: { iniciativaId_ciclo: { iniciativaId, ciclo } },
    });
    return linha
      ? { likes: linha.likes, dislikes: linha.dislikes, relevancia: linha.relevancia }
      : null;
  }

  async buscarPlacares(pares: ParIniciativaCiclo[]): Promise<Map<string, Placar>> {
    const resultado = new Map<string, Placar>();
    if (pares.length === 0) return resultado;
    const linhas = await this.host.client.placar.findMany({
      where: { OR: pares.map((p) => ({ iniciativaId: p.iniciativaId, ciclo: p.ciclo })) },
    });
    for (const linha of linhas) {
      resultado.set(linha.iniciativaId, {
        likes: linha.likes,
        dislikes: linha.dislikes,
        relevancia: linha.relevancia,
      });
    }
    return resultado;
  }
}

import { Body, Controller, HttpCode, Param, ParseUUIDPipe, Put } from '@nestjs/common';
import { ApiBearerAuth, ApiOperation, ApiResponse, ApiTags } from '@nestjs/swagger';
import type { CurrentUser } from '@/common/auth/current-user.interface';
import { GetCurrentUser } from '@/common/auth/decorators/current-user.decorator';
import { VotacaoFacade } from '../../application/votacao.facade';
import { RegistrarVotoDto, ResultadoVotoResponseDto } from './voto.dto';

@ApiTags('Votação')
@ApiBearerAuth()
@Controller({ path: 'iniciativas/:id/voto', version: '1' })
export class VotoController {
  constructor(private readonly votacao: VotacaoFacade) {}

  @Put()
  @HttpCode(200)
  @ApiOperation({
    summary: 'Registra, troca ou retira o voto do usuário',
    description:
      'Mesmo valor que o usuário já tem retira o voto; valor oposto troca. Só em Em votação.',
  })
  @ApiResponse({ status: 200, type: ResultadoVotoResponseDto })
  @ApiResponse({ status: 401, description: 'Não autenticado' })
  @ApiResponse({ status: 404, description: 'Iniciativa não encontrada' })
  @ApiResponse({ status: 409, description: 'A votação desta iniciativa foi encerrada' })
  @ApiResponse({ status: 422, description: 'Valor do voto inválido' })
  votar(
    @Param('id', new ParseUUIDPipe()) id: string,
    @Body() body: RegistrarVotoDto,
    @GetCurrentUser() user: CurrentUser,
  ): Promise<ResultadoVotoResponseDto> {
    return this.votacao.votar({ iniciativaId: id, usuarioId: user.userId, valor: body.valor });
  }
}
