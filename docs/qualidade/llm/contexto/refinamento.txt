import type { ComentarioPersistido } from '../../domain/ports/comentario.repository';
import type { AutorInfo } from '../ports/autores.port';

export type PerfilAutor = 'ALUNO' | 'PROFESSOR' | 'COMITE' | 'ADMIN';

export interface ComentarioDto {
  id: string;
  ciclo: number;
  autor: { id: string; nome: string; perfil: PerfilAutor; ehAutorDaIniciativa: boolean };
  texto: string;
  /** ISO 8601 */
  criadoEm: string;
}

export interface PaginaDeComentariosDto {
  items: ComentarioDto[];
  meta: { page: number; limit: number; total: number; totalPages: number };
}

export const LIMITE_PADRAO_COMENTARIOS = 50;

export function paraComentarioDto(
  comentario: ComentarioPersistido,
  autores: Map<string, AutorInfo>,
  autorDaIniciativaId: string,
): ComentarioDto {
  const autor = autores.get(comentario.autorId);
  return {
    id: comentario.id,
    ciclo: comentario.ciclo,
    autor: {
      id: comentario.autorId,
      nome: autor?.nome ?? 'Usuário desconhecido',
      perfil: (autor?.perfil as PerfilAutor | undefined) ?? 'ALUNO',
      ehAutorDaIniciativa: comentario.autorId === autorDaIniciativaId,
    },
    texto: comentario.texto,
    criadoEm: comentario.criadoEm.toISOString(),
  };
}

// ===== src/modules/refinamento/application/queries/listar-comentarios.ts =====
import { IniciativaNaoEncontradaError } from '../../domain/errors/refinamento.errors';
import type { ComentarioRepository } from '../../domain/ports/comentario.repository';
import {
  LIMITE_PADRAO_COMENTARIOS,
  type PaginaDeComentariosDto,
  paraComentarioDto,
} from '../dto/comentario.dto';
import type { AutoresPort } from '../ports/autores.port';
import type { IniciativaPort } from '../ports/iniciativa.port';

export interface ListarComentariosInput {
  iniciativaId: string;
  page?: number;
  limit?: number;
}

export class ListarComentarios {
  constructor(
    private readonly comentarios: ComentarioRepository,
    private readonly iniciativas: IniciativaPort,
    private readonly autores: AutoresPort,
  ) {}

  async executar(input: ListarComentariosInput): Promise<PaginaDeComentariosDto> {
    const page = input.page ?? 1;
    const limit = input.limit ?? LIMITE_PADRAO_COMENTARIOS;
    const estado = await this.iniciativas.obterEstado(input.iniciativaId);
    if (!estado) throw new IniciativaNaoEncontradaError();

    const { items, total } = await this.comentarios.listarPorIniciativa(
      estado.id,
      (page - 1) * limit,
      limit,
    );
    const autores = await this.autores.mapaPorIds([...new Set(items.map((i) => i.autorId))]);
    return {
      items: items.map((item) => paraComentarioDto(item, autores, estado.autorId)),
      meta: { page, limit, total, totalPages: Math.ceil(total / limit) },
    };
  }
}

// ===== src/modules/refinamento/application/use-cases/comentar-iniciativa.ts =====
import type { Clock } from '@/shared/domain/clock';
import { Comentario } from '../../domain/comentario';
import {
  FaseDeSugestoesEncerradaError,
  IniciativaNaoEncontradaError,
} from '../../domain/errors/refinamento.errors';
import type { ComentarioRepository } from '../../domain/ports/comentario.repository';
import { type ComentarioDto, paraComentarioDto } from '../dto/comentario.dto';
import type { AutoresPort } from '../ports/autores.port';
import type { IniciativaPort } from '../ports/iniciativa.port';

export interface ComentarIniciativaInput {
  iniciativaId: string;
  autorId: string;
  texto: string;
}

const STATUS_REFINAMENTO = 'EM_ANDAMENTO';

/** RN-05: qualquer usuário autenticado, inclusive o autor, comenta só em Em andamento. */
export class ComentarIniciativa {
  constructor(
    private readonly comentarios: ComentarioRepository,
    private readonly iniciativas: IniciativaPort,
    private readonly autores: AutoresPort,
    private readonly clock: Clock,
  ) {}

  async executar(input: ComentarIniciativaInput): Promise<ComentarioDto> {
    const texto = Comentario.validarTexto(input.texto);
    const estado = await this.iniciativas.obterEstado(input.iniciativaId);
    if (!estado) throw new IniciativaNaoEncontradaError();
    if (estado.status !== STATUS_REFINAMENTO) throw new FaseDeSugestoesEncerradaError();

    const comentario = Comentario.criar({
      iniciativaId: estado.id,
      ciclo: estado.ciclo,
      autorId: input.autorId,
      texto,
      criadoEm: this.clock.now(),
    });
    const salvo = await this.comentarios.salvar(comentario);
    const autores = await this.autores.mapaPorIds([salvo.autorId]);
    return paraComentarioDto(salvo, autores, estado.autorId);
  }
}

import { ValidacaoError } from '@/shared/domain/domain-error';

export const TEXTO_MAXIMO = 1000;

export const MENSAGEM_TEXTO_VAZIO = 'Escreva sua sugestão antes de enviar.';
export const MENSAGEM_TEXTO_LONGO = `A sugestão pode ter no máximo ${TEXTO_MAXIMO} caracteres.`;

export interface DadosComentario {
  iniciativaId: string;
  ciclo: number;
  autorId: string;
  texto: string;
  criadoEm: Date;
}

/** Comentário (sugestão) ainda não persistido. O texto é aparado e validado (RN-05). */
export class Comentario {
  readonly iniciativaId: string;
  readonly ciclo: number;
  readonly autorId: string;
  readonly texto: string;
  readonly criadoEm: Date;

  private constructor(dados: DadosComentario) {
    this.iniciativaId = dados.iniciativaId;
    this.ciclo = dados.ciclo;
    this.autorId = dados.autorId;
    this.texto = dados.texto;
    this.criadoEm = dados.criadoEm;
  }

  static criar(dados: DadosComentario): Comentario {
    return new Comentario({ ...dados, texto: Comentario.validarTexto(dados.texto) });
  }

  /** Devolve o texto aparado ou lança `ValidacaoError({ texto })`. */
  static validarTexto(texto: string): string {
    const aparado = texto.trim();
    if (aparado.length === 0) {
      throw new ValidacaoError({ texto: MENSAGEM_TEXTO_VAZIO });
    }
    if (aparado.length > TEXTO_MAXIMO) {
      throw new ValidacaoError({ texto: MENSAGEM_TEXTO_LONGO });
    }
    return aparado;
  }
}

// ===== src/modules/refinamento/infrastructure/persistence/prisma-comentario.repository.ts =====
import { Injectable } from '@nestjs/common';
import { PrismaTransactionHost } from '@/shared/infrastructure/prisma-transaction-host';
import type { Comentario } from '../../domain/comentario';
import type {
  ComentarioPersistido,
  ComentarioRepository,
  PaginaDeComentarios,
  ParIniciativaCiclo,
} from '../../domain/ports/comentario.repository';

@Injectable()
export class PrismaComentarioRepository implements ComentarioRepository {
  constructor(private readonly host: PrismaTransactionHost) {}

  async salvar(comentario: Comentario): Promise<ComentarioPersistido> {
    return this.host.client.comentario.create({
      data: {
        iniciativaId: comentario.iniciativaId,
        ciclo: comentario.ciclo,
        autorId: comentario.autorId,
        texto: comentario.texto,
        criadoEm: comentario.criadoEm,
      },
    });
  }

  async listarPorIniciativa(
    iniciativaId: string,
    skip: number,
    take: number,
  ): Promise<PaginaDeComentarios> {
    const client = this.host.client;
    const [items, total] = await Promise.all([
      client.comentario.findMany({
        where: { iniciativaId },
        orderBy: [{ ciclo: 'desc' }, { criadoEm: 'asc' }, { id: 'asc' }],
        skip,
        take,
      }),
      client.comentario.count({ where: { iniciativaId } }),
    ]);
    return { items, total };
  }

  async contarPorCiclo(pares: ParIniciativaCiclo[]): Promise<Map<string, number>> {
    const contagens = new Map<string, number>();
    if (pares.length === 0) return contagens;
    const grupos = await this.host.client.comentario.groupBy({
      by: ['iniciativaId', 'ciclo'],
      where: { OR: pares.map((p) => ({ iniciativaId: p.iniciativaId, ciclo: p.ciclo })) },
      _count: { _all: true },
    });
    for (const grupo of grupos) {
      const atual = contagens.get(grupo.iniciativaId) ?? 0;
      contagens.set(grupo.iniciativaId, atual + grupo._count._all);
    }
    return contagens;
  }
}

import { Body, Controller, Get, HttpCode, Param, ParseUUIDPipe, Post, Query } from '@nestjs/common';
import { ApiBearerAuth, ApiOperation, ApiResponse, ApiTags } from '@nestjs/swagger';
import type { CurrentUser } from '@/common/auth/current-user.interface';
import { GetCurrentUser } from '@/common/auth/decorators/current-user.decorator';
import { RefinamentoFacade } from '../../application/refinamento.facade';
import {
  ComentarioResponseDto,
  CriarComentarioDto,
  ListarComentariosQueryDto,
  PaginaDeComentariosResponseDto,
} from './comentario.dto';

@ApiTags('Refinamento')
@ApiBearerAuth()
@Controller({ path: 'iniciativas/:id/comentarios', version: '1' })
export class ComentariosController {
  constructor(private readonly refinamento: RefinamentoFacade) {}

  @Get()
  @ApiOperation({
    summary: 'Lista os comentários da iniciativa',
    description:
      'Ciclo mais recente primeiro e, dentro do ciclo, do mais antigo para o mais novo. Limite padrão: 50.',
  })
  @ApiResponse({ status: 200, type: PaginaDeComentariosResponseDto })
  @ApiResponse({ status: 401, description: 'Não autenticado' })
  @ApiResponse({ status: 404, description: 'Iniciativa não encontrada' })
  @ApiResponse({ status: 422, description: 'Parâmetros de paginação inválidos' })
  listar(
    @Param('id', new ParseUUIDPipe()) id: string,
    @Query() query: ListarComentariosQueryDto,
  ): Promise<PaginaDeComentariosResponseDto> {
    return this.refinamento.listar({ iniciativaId: id, page: query.page, limit: query.limit });
  }

  @Post()
  @HttpCode(201)
  @ApiOperation({
    summary: 'Comenta (sugere) na iniciativa',
    description: 'Só enquanto a iniciativa está Em andamento. Qualquer usuário, inclusive o autor.',
  })
  @ApiResponse({ status: 201, type: ComentarioResponseDto })
  @ApiResponse({ status: 401, description: 'Não autenticado' })
  @ApiResponse({ status: 404, description: 'Iniciativa não encontrada' })
  @ApiResponse({ status: 409, description: 'A fase de sugestões terminou' })
  @ApiResponse({ status: 422, description: 'Texto vazio ou com mais de 1000 caracteres' })
  comentar(
    @Param('id', new ParseUUIDPipe()) id: string,
    @Body() body: CriarComentarioDto,
    @GetCurrentUser() user: CurrentUser,
  ): Promise<ComentarioResponseDto> {
    return this.refinamento.comentar({
      iniciativaId: id,
      autorId: user.userId,
      texto: body.texto,
    });
  }
}
